#!/usr/bin/env python3
"""
project_restore.py

Restore a project from project_dump.txt generated by project_dump.py.

Usage:
    python project_restore.py

or

    python project_restore.py path/to/project_dump.txt

The project will be recreated in the folder where this script is executed.
"""

from pathlib import Path
import re
import sys

# ============================================================================
# Configuration
# ============================================================================

DEFAULT_DUMP_FILE = "project_dump.txt"

# Set to False if you don't want existing files overwritten.
OVERWRITE_EXISTING = True

# ============================================================================

FILE_MARKER = re.compile(r"^FILE:\s*(.+)$")


def restore_project(dump_file: Path, output_dir: Path):

    if not dump_file.exists():
        print(f"ERROR: Dump file not found: {dump_file}")
        return

    print("=" * 70)
    print("PROJECT RESTORE")
    print("=" * 70)
    print(f"Input : {dump_file}")
    print(f"Output: {output_dir}")
    print()

    restored = 0
    skipped = 0

    current_file = None
    current_lines = []
    collecting = False

    with open(dump_file, "r", encoding="utf-8", errors="ignore") as f:

        for line in f:

            line = line.rstrip("\n")

            # -----------------------------------------------------------------
            # Detect FILE:
            # -----------------------------------------------------------------
            match = FILE_MARKER.match(line)

            if match:

                # Save previous file
                if current_file is not None:

                    save_file(
                        output_dir,
                        current_file,
                        current_lines,
                    )

                    restored += 1

                current_file = Path(match.group(1).strip())
                current_lines = []
                collecting = False
                continue

            # Ignore separator lines
            if line.startswith("#") and len(line) > 20:
                collecting = True
                continue

            if current_file is None:
                continue

            if not collecting:
                continue

            current_lines.append(line)

    # Save last file
    if current_file is not None:

        save_file(
            output_dir,
            current_file,
            current_lines,
        )

        restored += 1

    print()
    print("=" * 70)
    print(f"Files Restored : {restored}")
    print(f"Files Skipped  : {skipped}")
    print("=" * 70)
    print("Done.")


def save_file(base: Path, relative: Path, lines):

    target = base / relative

    target.parent.mkdir(parents=True, exist_ok=True)

    if target.exists() and not OVERWRITE_EXISTING:
        print(f"Skip : {relative}")
        return

    # Remove trailing blank lines
    while lines and lines[-1] == "":
        lines.pop()

    with open(target, "w", encoding="utf-8") as f:
        f.write("\n".join(lines))

    print(f"Created : {relative}")


def main():

    if len(sys.argv) > 1:
        dump_file = Path(sys.argv[1])
    else:
        dump_file = Path(DEFAULT_DUMP_FILE)

    output_dir = Path.cwd()

    restore_project(
        dump_file=dump_file,
        output_dir=output_dir,
    )


if __name__ == "__main__":
    main()

## Architecture

### Component Layers

```
╔══════════════════════════════════════════════════════════════════╗
║  PRESENTATION LAYER                                              ║
║                                                                  ║
║   application/app.py  (Streamlit)                                ║
║   • LLM Configuration page — test & persist DLH credentials     ║
║   • Chat page — conversational NL2SQL with stage-pill trace      ║
║   • Session state stores processor & chat history               ║
╠══════════════════════════════════════════════════════════════════╣
║  ORCHESTRATION LAYER                                             ║
║                                                                  ║
║   ComplianceProcessor  (genai/core/processor.py)                 ║
║   • Builds AgentState from the user question                     ║
║   • Calls compiled LangGraph graph                               ║
║   • Maps final AgentState → response                             ║
╠══════════════════════════════════════════════════════════════════╣
║  PIPELINE LAYER  (LangGraph StateGraph)                          ║
║                                                                  ║
║   Node 1   IntentChecker    rule-based, no LLM                   ║
║   Node 2   SchemaChecker    rule-based, no LLM                   ║
║   Node 3   SQLWriterAgent   ★ only LLM call in the pipeline      ║
║   Node 4   SQLReviewer      rule-based, no LLM                   ║
║   Node 4.5 SQLValidator     sqlglot AST, no LLM                  ║
║   Node 5   ExecutorAgent    rule-based, no LLM                   ║
╠══════════════════════════════════════════════════════════════════╣
║  INFRASTRUCTURE LAYER                                            ║
║                                                                  ║
║   utils/config.py    Pydantic Settings — reads .env              ║
║   utils/logger.py    get_logger() — reads logging.toml           ║
║   utils/registry.py  YAML loader + SQLite helpers + DB registry  ║
║   utils/helpers.py   stateless stdlib helpers                    ║
║                                                                  ║
║   genai/config/*.toml   TOML config (no secrets)                 ║
║   genai/prompt/*.yaml   YAML prompt files (LLM wording)          ║
╠══════════════════════════════════════════════════════════════════╣
║  STORAGE LAYER                                                   ║
║                                                                  ║
║   data/comp_db/COMP_REPORTS.db   SQLite — read-only at runtime   ║
║   data/logs/compliance_bot.log   Rotating log file               ║
║   llm_config.json                Persisted LLM credentials       ║
╚══════════════════════════════════════════════════════════════════╝
```

### LangGraph Pipeline Flow

```
[START]
   │
   ▼
intent_checker  ──(intent_ok=False)──► clarify_node ──► [END]
   │ (intent_ok=True)
   ▼
schema_checker
   │
   ▼
sql_writer_agent  ◄──────────────────────────────────────────┐
   │                                                         │ retry
   ▼                                                         │
sql_reviewer ──(passed)──► sql_validator ────────────────────┤
   │                            │                            │
   │ (max retries / error)      │ (valid)    (invalid → retry)
   ▼                            ▼
executor_agent ◄──────── executor_agent ──► [END]
```

| Node | Module | LLM? | Role |
|---|---|---|---|
| `intent_checker` | `genai/nl2sql_agent/intent_checker.py` | ✗ | Rule-based intent gate + normalisation |
| `schema_checker` | `genai/nl2sql_agent/schema_checker.py` | ✗ | Live PRAGMA schema + TOML descriptions |
| `sql_writer_agent` | `genai/nl2sql_agent/sql_writer_agent.py` | ✓ | Single LLM call — SQL generation |
| `sql_reviewer` | `genai/nl2sql_agent/sql_reviewer.py` | ✗ | Safety + structural + table existence |
| `sql_validator` | `genai/nl2sql_agent/sql_validator.py` | ✗ | sqlglot AST + dialect check |
| `executor_agent` | `genai/nl2sql_agent/executor_agent.py` | ✗ | Read-only SQLite execution |

## Adding a New Database

**No Python code changes are needed.** Edit `genai/config/databases.toml` only:

```toml
[databases.MY_NEW_DB]
db_file     = "data/comp_db/MY_NEW_DB.db"
description = "Description of the new database."

[databases.MY_NEW_DB.tables.MY_TABLE]
description  = "Plain-English summary for the LLM prompt."
filter_query = "SELECT * FROM MY_TABLE WHERE BRANCH = '{branch}'"

[databases.MY_NEW_DB.tables.MY_TABLE.columns]
COL1 = "Human-readable column 1 description"
COL2 = "Human-readable column 2 description"

[databases.MY_NEW_DB.tables.MY_TABLE.few_shots]
examples = """
Question: Show all rows where COL1 is active.
Response:
<begin_sql_response>
SELECT * FROM MY_TABLE WHERE COL1 = 'active';
<end_sql_response>
"""
```

Restart Streamlit to pick up the changes.  
See [`docs/add-database.md`](docs/add-database.md) for full details.

## Design Principles

* **One LLM call per query** — only `SQLWriterAgent` calls the LLM; all other nodes are deterministic and fast.
* **YAML-first prompts** — all LLM wording lives in `genai/prompt/*.yaml`; change messages without touching Python.
* **Zero-code database adds** — add a new database by editing `databases.toml` only.
* **Read-only safety** — every SQL execution is wrapped in a `BEGIN DEFERRED` transaction that is unconditionally rolled back.
* **Defence in depth** — SQL passes through three independent validation layers (regex review → sqlglot AST → SQLite error handling) before results are returned.
* **Persistent LLM config** — credentials entered in the UI are saved to `llm_config.json` and survive Streamlit restarts.

## Troubleshooting

### "Please configure the LLM first!"
The pipeline is not initialised. Go to **LLM Configuration** in the sidebar and click **Test & Save**.

### Connection test fails
1. Check `MODEL__DLH_BASE_URL` is correct.
2. Confirm the inference server is reachable:
   ```bash
   curl http://10.190.236.15:9000/api/v1/models
   ```
3. Verify the API key is set correctly (even `EMPTY` must be the exact value expected by the server).

### `ModuleNotFoundError`
Run from the workspace root, not from inside `application/`:
```bash
# ✗ wrong
cd dlh_ai/application && streamlit run app.py

# ✓ correct
streamlit run dlh_ai/application/app.py
```

### Database not found
Run the seed script first:
```bash
python dlh_ai/data/seed_synthetic_data.py
```

CONFIG_FILE = _PROJECT_ROOT / "llm_config.json"  # File where settings are persisted

if str(_PROJECT_ROOT) not in sys.path:
    sys.path.insert(0, str(_PROJECT_ROOT))

from genai.core.model import ModelLoaderService

LOGGER = get_logger("compliance_bot.streamlit")

# --- PERSISTENCE HELPERS ---
def save_config_to_disk(model_name, base_url, api_key):
    """Saves configuration to a local JSON file."""
    try:
        config = {
            "model_name": model_name,
            "base_url": base_url,
            "api_key": api_key
        }
        with open(CONFIG_FILE, "w") as f:
            json.dump(config, f)
        LOGGER.info("Configuration saved to disk successfully.")
    except Exception as e:
        LOGGER.error(f"Failed to save config to disk: {e}")

def load_config_from_disk():
    """Loads configuration from the local JSON file."""
    try:
        if CONFIG_FILE.exists():
            with open(CONFIG_FILE, "r") as f:
                return json.load(f)
    except Exception as e:
        LOGGER.error(f"Failed to load config from disk: {e}")
    return None

# --- FONT LOADING HELPER ---
def get_font_base64(font_path):
    """Encodes local font file to base64 for CSS injection"""
    try:
        with open(font_path, "rb") as f:
            data = f.read()
        return base64.b64encode(data).decode()
    except Exception as e:
        LOGGER.error(f"Could not load font at {font_path}: {e}")
        return None

font_bold = get_font_base64(font_dir / "Urbanist-Bold.ttf")

st.set_page_config(

def ensure_database(db_path: Path) -> None:
    """
    Create the parent directory and SQLite database file if they do not exist.

    This is a no-op when both already exist — SQLite creates the file on
    first ``connect()`` so the function just guarantees the directory is there.

    Args:
        db_path: Absolute or relative path to the ``.db`` file.
    """
    db_path.parent.mkdir(parents=True, exist_ok=True)
    existed = db_path.exists()
    sqlite3.connect(str(db_path)).close()
    if not existed:
        print(f"[create] Database file created: {db_path}")
    else:
        print(f"[ok]     Database file exists:  {db_path}")


def ensure_table(conn: sqlite3.Connection) -> None:
    """
    Create ``COMP_TBL`` with the full 25-column schema if it does not exist.

    If the table already exists but is missing required columns (stale schema),
    it is dropped and recreated so the seeder can always run safely.

    Args:
        conn: Open SQLite connection to the target database.
    """
    table_exists = conn.execute(
        "SELECT name FROM sqlite_master WHERE type='table' AND name='COMP_TBL'"
    ).fetchone()

    if table_exists:
        existing_cols: set[str] = {
            row[1] for row in conn.execute("PRAGMA table_info(COMP_TBL)").fetchall()
        }
        if not _REQUIRED_COLUMNS.issubset(existing_cols):
            conn.execute("DROP TABLE COMP_TBL")
            conn.commit()
            print("[migrate] COMP_TBL schema was stale — dropped and will be recreated.")
            table_exists = None  # fall through to CREATE

    if not table_exists:
        conn.execute(_CREATE_TABLE_SQL)
        conn.commit()
        print("[create] COMP_TBL created.")
    else:
        print("[ok]     COMP_TBL already exists with correct schema.")


def reset_table(conn: sqlite3.Connection) -> None:
    """
    Delete every row from ``COMP_TBL``.

    Args:
        conn: Open SQLite connection to the target database.
    """
    deleted: int = conn.execute("DELETE FROM COMP_TBL").rowcount
    conn.commit()
    print(f"[reset]  Deleted {deleted:,} existing rows from COMP_TBL.")


def _rand_date(start: date, end: date) -> date:
    """Return a random date between *start* and *end* (inclusive)."""
    return start + timedelta(days=random.randint(0, (end - start).days))


def _fmt(d: date) -> str:
    """Format a date as ISO-8601 string (YYYY-MM-DD)."""
    return d.strftime("%Y-%m-%d")


def _rand_rate(base: float) -> float:
    """Return a rate within ±0.75 pp of *base*, rounded to 2 dp."""
    return round(base + random.uniform(-0.75, 0.75), 2)


def _make_row(i: int) -> tuple:  # type: ignore[type-arg]
    """Generate one synthetic COMP_TBL row."""
    circle_name, circle_cd = random.choice(_CIRCLES)
    branch_name            = random.choice(_BRANCHES[circle_cd])
    prdct_id, product_name = random.choice(_PRODUCTS)

    age: int     = random.randint(18, 85)
    sr_flag: int = 1 if age >= 60 else 0
    dob: str     = _fmt(_rand_date(date(2025 - age - 1, 1, 1), date(2025 - age, 12, 31)))

    kyc: str        = "Y" if random.random() < (0.80 if sr_flag else 0.90) else "N"
    balance: float  = round(random.lognormvariate(11.5, 1.2), 2)
    balance         = max(10_000.0, min(5_000_000.0, balance))

    open_dt: date   = _rand_date(date(2005, 1, 1), date(2024, 6, 30))
    tenor_yr: int   = random.choice([1, 2, 3, 5, 7, 10, 15])
    tenor_day: int  = random.randint(0, 364)
    mat_dt_d: date  = open_dt + timedelta(days=tenor_yr * 365 + tenor_day)
    mat_dt: str     = _fmt(mat_dt_d)

    is_mod: str  = "YES" if random.random() < 0.15 else "NO"
    mtrty_dt: str = _fmt(mat_dt_d + timedelta(days=random.randint(-30, 30))) if is_mod == "YES" else mat_dt

    base_rate: float = 7.0 if sr_flag else 6.5
    new_rate: float  = _rand_rate(base_rate)
    exp_rate: float  = _rand_rate(base_rate + 0.25)
    pf_fac: str      = "Y" if random.random() < 0.30 else "N"
    brcd: str        = f"{random.randint(1, 999):03d}"
    acct_brnch: str  = f"M{random.randint(1, 99):03d}"

    return (
        brcd,                        # BRCD
        f"C{i:06d}",                 # CUST_NBR
        str(random.randint(1, 9)),   # CUST_NBR_CHKDGT
        f"A{i:06d}",                 # ACCT_NBR
        str(random.randint(1, 9)),   # ACCT_NBR_CHKDGT
        acct_brnch,                  # ACCT_BRNCH
        dob,                         # BRTH_DT
        age,                         # AGE_YRS
        sr_flag,                     # SR_CTZN_ACCT_FLG
        kyc,                         # KYC_FLG
        balance,                     # END_OF_DAY_BAL
        _fmt(open_dt),               # ACCT_OPN_DT
        mtrty_dt,                    # MTRTY_DT
        tenor_yr,                    # tenor_yr
        tenor_day,                   # tenor_day
        mat_dt,                      # MAT_DT
        new_rate,                    # NEW_EFFCTV_RT
        prdct_id,                    # PRDCT_ID
        pf_fac,                      # PF_FAC_REPT
        product_name,                # PRODUCT_NAME
        branch_name,                 # BRANCHNAME
        circle_name,                 # CIRCLENAME
        circle_cd,                   # CIRCLECD
        exp_rate,                    # EXPECTED_EFFCTV_RT
        is_mod,                      # ISMOD
    )


def generate_rows(count: int, start_i: int) -> list[tuple]:  # type: ignore[type-arg]
    """
    Generate *count* synthetic row tuples starting from index *start_i*.

    ``start_i`` is used for ``CUST_NBR`` / ``ACCT_NBR`` sequential IDs so
    appending rows never collides with existing ones.

    Args:
        count:   Number of rows to generate.
        start_i: Index of the first new row (1-based).

    Returns:
        List of row tuples ready for ``executemany``.
    """
    return [_make_row(start_i + i) for i in range(count)]


def insert_rows(conn: sqlite3.Connection, rows: list[tuple]) -> None:  # type: ignore[type-arg]
    """
    Bulk-insert *rows* into ``COMP_TBL`` using a single ``executemany``.

    Args:
        conn: Open SQLite connection (must already have COMP_TBL).
        rows: Row tuples as returned by :func:`generate_rows`.
    """
    conn.executemany(_INSERT_SQL, rows)
    conn.commit()
    print(f"[insert] {len(rows):,} rows inserted into COMP_TBL.")


def print_summary(conn: sqlite3.Connection, db_path: Path, inserted: int) -> None:
    """
    Print a statistics summary after the insert.

    Args:
        conn:     Open SQLite connection.
        db_path:  Path to the database file (for display only).
        inserted: Number of rows that were just inserted.
    """
    total: int   = conn.execute("SELECT COUNT(*) FROM COMP_TBL").fetchone()[0]
    sr: int      = conn.execute("SELECT COUNT(*) FROM COMP_TBL WHERE SR_CTZN_ACCT_FLG = 1").fetchone()[0]
    kyc_n: int   = conn.execute("SELECT COUNT(*) FROM COMP_TBL WHERE KYC_FLG = 'N'").fetchone()[0]
    circles: list[tuple[str, int]] = conn.execute(
        "SELECT CIRCLENAME, COUNT(*) FROM COMP_TBL GROUP BY CIRCLENAME ORDER BY 2 DESC LIMIT 5"
    ).fetchall()

    print(f"\n{'='*55}")
    print(f"  Database : {db_path}")
    print(f"  Inserted : {inserted:,} rows")
    print(f"  Total    : {total:,} rows in COMP_TBL")
    print(f"{'='*55}")
    print(f"  Senior citizens  (SR_CTZN_ACCT_FLG=1) : {sr:,}")
    print(f"  KYC non-compliant (KYC_FLG='N')        : {kyc_n:,}")
    print("\n  Top 5 circles by row count:")
    for name, cnt in circles:
        print(f"    {name:<22} {cnt:>5}")
    print(f"{'='*55}\n")


def seed(db_path: Path, count: int, reset: bool) -> None:
    """
    Full end-to-end pipeline:

    1. :func:`ensure_database` — create ``.db`` file and parent dirs if absent.
    2. :func:`ensure_table`    — create ``COMP_TBL`` if absent or stale.
    3. :func:`reset_table`     — (only when ``reset=True``) delete all rows.
    4. :func:`generate_rows`   — build synthetic row tuples in memory.
    5. :func:`insert_rows`     — bulk-insert into the database.
    6. :func:`print_summary`   — display post-insert statistics.

    Args:
        db_path: Path to the SQLite ``.db`` file.
        count:   Number of rows to insert (0 = create DB/table only).
        reset:   When ``True``, delete all existing rows before inserting.
    """
    ensure_database(db_path)

    conn = sqlite3.connect(str(db_path))

    try:
        ensure_table(conn)

        if reset:
            reset_table(conn)

        if count == 0:
            print("[info]   --count 0: skipping row insert.")
            print_summary(conn, db_path, inserted=0)
            return

        start_i: int = conn.execute("SELECT COUNT(*) FROM COMP_TBL").fetchone()[0] + 1
        rows = generate_rows(count, start_i)

        insert_rows(conn, rows)

        print_summary(conn, db_path, inserted=count)

    finally:
        conn.close()


def _parse_args() -> argparse.Namespace:
    p = argparse.ArgumentParser(
        description=(
            "Create the SQLite database + COMP_TBL schema and insert synthetic data.\n"
            "Safe to re-run: creates only what is missing."
        ),
        formatter_class=argparse.RawDescriptionHelpFormatter,
    )
    p.add_argument(
        "--count", type=int, default=2000,
        help="Number of rows to insert (default: 2000). Use 0 to create DB/table only.",
    )
    p.add_argument(
        "--reset", action="store_true",
        help="Delete all existing rows before inserting.",
    )
    p.add_argument(
        "--db", type=Path,
        default=Path(__file__).resolve().parent / "comp_db" / "COMP_REPORTS.db",
        help="Path to the SQLite database file.",
    )
    p.add_argument(
        "--seed", type=int, default=42,
        help="Random seed for reproducibility (default: 42).",
    )
    return p.parse_args()


if __name__ == "__main__":
    args = _parse_args()
    random.seed(args.seed)
    seed(db_path=args.db, count=args.count, reset=args.reset)

# Adding a New Database

This page walks you through connecting a second (or third) SQLite database to the pipeline with **minimal code changes** — the architecture is designed so that adding a database is almost entirely a configuration task.

**Total files to edit: 2** (plus your `.db` file itself).

---

## Overview

The pipeline discovers databases at runtime by:

1. `SchemaChecker` scanning the `data_root` folder for `*.db` files and scoring each one against the question.
2. `SQLWriterAgent` reading table metadata (descriptions, column definitions, few-shots) from `genai/config/databases.toml`.

You do **not** need to touch any Python files.

---

## Step 1 — Place the Database File

Put the SQLite database file anywhere under `dlh_ai/data/`:

```
dlh_ai/data/
├── comp_db/
│   └── COMP_REPORTS.db          ← existing
└── my_new_db/
    └── MY_NEW_DATABASE.db       ← new file
```

> The exact sub-folder name does not matter — `SchemaChecker` uses `data_root.glob("**/*.db")`.

If you need to create the database from a CSV or schema script, run that first and place the `.db` file here.

---

## Step 2 — Register in `databases.toml`

Open `dlh_ai/genai/config/databases.toml` and add a new `[databases.<key>]` block:

```toml
[databases.COMP_REPORTS]
db_file     = "data/comp_db/COMP_REPORTS.db"
description = "Banking compliance reports database for term-deposit products."

[databases.COMP_REPORTS.tables.COMP_TBL]
description  = "..."
filter_query = "SELECT * FROM COMP_TBL WHERE ACCT_BRNCH IN ({branch_code})"

[databases.COMP_REPORTS.tables.COMP_TBL.columns]
BRCD = "BRANCH CODE OR BRANCH NUMBER"
# ... existing columns ...


[databases.MY_NEW_DATABASE]
db_file     = "data/my_new_db/MY_NEW_DATABASE.db"
description = "Short description of what this database contains."

[databases.MY_NEW_DATABASE.tables.MY_TABLE]
description  = "What this table represents."
filter_query = "SELECT * FROM MY_TABLE WHERE BRANCH_ID = '{branch_id}'"

[databases.MY_NEW_DATABASE.tables.MY_TABLE.columns]
BRANCH_ID    = "Unique identifier for the branch"
CUSTOMER_ID  = "Unique customer identifier"
PRODUCT_TYPE = "Type of financial product"
BALANCE      = "Current account balance in INR"
STATUS       = "Account status: ACTIVE, CLOSED, DORMANT"

[databases.MY_NEW_DATABASE.tables.MY_TABLE.few_shots]
examples = """
Question: Show all active accounts in MY_TABLE.
Response:
<begin_sql_response>
SELECT * FROM MY_TABLE WHERE STATUS = 'ACTIVE';
<end_sql_response>

Question: Count accounts by product type.
Response:
<begin_sql_response>
SELECT PRODUCT_TYPE, COUNT(*) AS cnt FROM MY_TABLE GROUP BY PRODUCT_TYPE ORDER BY cnt DESC;
<end_sql_response>
"""
```

### Field reference

| Field | Required | Description |
|-------|----------|-------------|
| `db_file` | ✓ | Relative path to the `.db` file from the **project root** (`dlh_ai/`) |
| `description` | recommended | Injected into the LLM prompt as context |
| `tables.<NAME>.description` | recommended | Per-table description in the LLM prompt |
| `tables.<NAME>.filter_query` | optional | Template for branch-filtered queries |
| `tables.<NAME>.columns.<COL>` | recommended | Business description of each column |
| `tables.<NAME>.few_shots.examples` | strongly recommended | NL→SQL examples for this table |

---

## Step 3 — (Optional) Change the Default Database

By default the app uses the database configured under `DATABASE__DB_FOLDER` / `DATABASE__DB_NAME` in `.env` (or `AppSettings` defaults). If you want **this** new database to be the default:

```dotenv
# .env
DATABASE__DB_FOLDER=data/my_new_db
DATABASE__DB_NAME=MY_NEW_DATABASE.db
```

If you want to keep `COMP_REPORTS.db` as the default but also support the new one, **no `.env` change is needed** — `SchemaChecker` automatically routes questions to the best-matching database based on keyword overlap.

---

## Step 4 — Verify

```python
# Run from workspace root
python -c "
import sys; sys.path.insert(0, 'dlh_ai')
from utils.registry import _build_db_registry
reg = _build_db_registry()
print(list(reg.keys()))
"
```

You should see `MY_TABLE` (or whatever your table name is) alongside `COMP_TBL`.

---

## Step 5 — Test

Restart Streamlit, then ask a question specific to the new table and check the **Agent Trace** in the result — you should see `schema_checker:db=MY_NEW_DATABASE.db`.

---

## Troubleshooting

| Symptom | Likely cause | Fix |
|---------|-------------|-----|
| SchemaChecker still uses old DB | Keyword scores tied | Add more specific question keywords in `few_shots.examples` |
| `MY_TABLE` not in registry | Typo in `databases.toml` table key | Key must exactly match the SQLite table name |
| `FileNotFoundError` in registry | Wrong `db_file` path | Path is relative to `dlh_ai/` project root, not to `genai/config/` |
| SQL always wrong for new table | No few-shots | Add at least 3 diverse `few_shots.examples` |
| Column names rejected by reviewer | Table not in `sqlite_master` | Ensure the `.db` file exists and the table was created |

---

## Summary

| # | File | What to add |
|---|------|-------------|
| 1 | `dlh_ai/data/<folder>/<NAME>.db` | The SQLite database file itself |
| 2 | `dlh_ai/genai/config/databases.toml` | One `[databases.<KEY>]` block per database, one `[tables.<NAME>]` block per table |

That's it. No Python changes required.

---

## Replacing the Default Table Entirely

If your new table should be the **only** table the pipeline ever queries:

1. Set `DATABASE__DB_NAME` in `.env` to your new `.db` file.
2. Set `COMPLIANCE_TABLE_NAME` in `dlh_ai/utils/config.py`:

```python
COMPLIANCE_TABLE_NAME: str = "MY_TABLE"   # was "COMP_TBL"
```

3. Update `SQLWriterAgent` default in `dlh_ai/genai/nl2sql_agent/sql_writer_agent.py`:

```python
class SQLWriterAgent:
    def __init__(
        self,
        model_service: ModelLoaderService,
        table_name: str = "MY_TABLE",    # was COMPLIANCE_TABLE_NAME
    ) -> None:
```

# Agent Nodes

The NL-to-SQL pipeline is a six-node **LangGraph StateGraph**.  
Every node receives the full `AgentState` dict, reads the keys it needs, and returns a **partial** dict containing only the keys it owns.

---

## Shared State — `AgentState`

`genai/nl2sql_agent/state.py`

```python
class AgentState(TypedDict, total=False):
    # ── Input ──────────────────────────────────────────────
    question:             str          # raw user question
    question_id:          str | None   # optional idempotency key

    # ── Node 1 outputs ─────────────────────────────────────
    interpreted_question: str          # normalised question with trailing ?
    intent_ok:            bool         # True = proceed, False = clarify
    clarification_needed: str | None   # message shown to the user when False

    # ── Node 2 outputs ─────────────────────────────────────
    db_path:              str          # absolute path to the chosen .db file
    schema_context:       str          # Markdown schema block injected into LLM prompt

    # ── Node 3 outputs ─────────────────────────────────────
    sql_query:            str          # raw SQL returned by the LLM

    # ── Node 4 outputs ─────────────────────────────────────
    sql_review_passed:    bool
    review_feedback:      str | None   # sent back to Node 3 on retry
    retry_count:          int

    # ── Node 4.5 outputs ───────────────────────────────────
    sqlglot_valid:        bool
    sqlglot_error:        str | None

    # ── Node 5 outputs ─────────────────────────────────────
    result_rows:          list[dict[str, object]]
    result_columns:       list[str]
    result_summary:       str

    # ── Shared bookkeeping ─────────────────────────────────
    error:                str | None
    stages:               list[str]    # append-only audit trail (LangGraph reducer)
```

The `stages` field uses `Annotated[list[str], operator.add]` so LangGraph **appends** each node's entries rather than overwriting the whole list.

---

## Node 1 — IntentChecker

**File:** `genai/nl2sql_agent/intent_checker.py`  
**Prompt config:** `genai/prompt/intent_checker.yaml`  
**Type:** Rule-based (no LLM call)

### What it does

Runs four sequential checks. The first check that fails returns `intent_ok=False` immediately; all four must pass for the pipeline to continue.

| Check | Rule | Failure example |
|-------|------|----------------|
| **Empty** | question is blank after strip | `""` or `"   "` |
| **Disallowed action** | token matches DROP / DELETE / INSERT / UPDATE / TRUNCATE / ALTER / CREATE / REPLACE / PRAGMA | `"delete all accounts"` |
| **Too short** | fewer than 3 meaningful tokens **and** no metric term | `"pending"` |
| **Vague / ambiguous** | phrase is in `_VAGUE_PHRASES` set, or contains an ambiguous pronoun with no metric term | `"give me data"`, `"please verify this"` |

If the question contains any of the 24 metric terms (`age`, `kyc`, `balance`, `maturity`, `account`, `count`, `sum`, `top` …) the too-short check is skipped, allowing short but specific questions like `"show KYC"`.

### Outputs written

```
interpreted_question   — stripped + trailing "?" added
intent_ok              — True / False
clarification_needed   — None or YAML message string
stages                 — ["intent_checker:ok"] or ["intent_checker:too_short"] etc.
```

### Routing

```
intent_ok=True  → Node 2 (SchemaChecker)
intent_ok=False → clarify_node → END
```

---

## Node 2 — SchemaChecker

**File:** `genai/nl2sql_agent/schema_checker.py`  
**Prompt config:** `genai/prompt/schema_checker.yaml`  
**Type:** Rule-based (no LLM call)

### What it does

1. Scans `data_root` (the DB folder) for every `*.db` file.
2. Scores each file by keyword overlap between the question and table/column names.
3. Selects the highest-scoring `.db` file.
4. Builds a **Markdown schema context** string containing:
   - Table names with column types (from `PRAGMA table_info`)
   - Sample rows (up to `schema.sample_row_limit` from `schema_checker.yaml`, default 3)
   - Column business descriptions from `databases.toml`

### Scoring algorithm

```
score = 1  if db_filename_stem is in the question (case-insensitive)
     += 2  for each table name found in the question
     += 1  for each column name found in the question
```

### Outputs written

```
db_path        — absolute path to the winning .db file
schema_context — Markdown string (injected verbatim into the LLM prompt)
stages         — ["schema_checker:db=COMP_REPORTS.db"]
```

---

## Node 3 — SQLWriterAgent ★

**File:** `genai/nl2sql_agent/sql_writer_agent.py`  
**Prompt config:** `genai/prompt/sql_writer.yaml`  
**Type:** **LLM call** (only node that calls the LLM)

### What it does

1. `ContextBuilder.build()` assembles the full prompt from:
   - System role (from `sql_writer.yaml`)
   - Numbered rules list
   - Table description + column block (from `databases.toml`)
   - Live schema context (from Node 2)
   - Few-shot examples (from `databases.toml`)
   - Reviewer feedback (present only on retries)
2. Calls `ModelLoaderService.generate(prompt)`.
3. Extracts SQL from between `<begin_sql_response>` … `<end_sql_response>` tags.

### Retry behaviour

On retry (coming back from Node 4 or 4.5), `review_feedback` is non-null and gets injected as a `[REVIEWER FEEDBACK]` block at the bottom of the prompt, so the LLM can see exactly what was wrong.

### Outputs written

```
sql_query    — extracted SQL string (or "" if extraction failed)
retry_count  — unchanged (passed through)
stages       — ["sql_writer_agent:llm_generated retry=0"]
error        — set only if no SQL tags found in output
```

---

## Node 4 — SQLReviewer

**File:** `genai/nl2sql_agent/sql_reviewer.py`  
**Prompt config:** `genai/prompt/sql_review.yaml`  
**Type:** Rule-based (no LLM call)

### What it does

Six checks in order:

| # | Check | Action on fail |
|---|-------|---------------|
| 1 | `retry_count >= max_retries` (default 3) | Sets `error`, routes to Executor to surface the message |
| 2 | SQL is empty | Increments retry, routes back to Writer |
| 3 | Contains disallowed DML/DDL keyword | Increments retry, routes back to Writer |
| 4 | Does not start with `SELECT` | Increments retry, routes back to Writer |
| 5 | Missing `FROM` clause | Increments retry, routes back to Writer |
| 6 | Referenced tables don't exist in the `.db` file | Increments retry, routes back to Writer |

### Routing

```
error set        → Executor (surfaces the error message to user)
passed=True      → Node 4.5 (SQLValidator)
passed=False     → Node 3 (SQLWriter, retry)
```

---

## Node 4.5 — SQLValidator

**File:** `genai/nl2sql_agent/sql_validator.py`  
**Type:** Rule-based — uses **sqlglot** AST parser (no LLM call)

### What it does

Three checks:

1. **Parse validity** — `sqlglot.parse()` must succeed without raising `ParseError`.
2. **Statement type** — parsed AST root must be a `Select` node.
3. **Dialect canonicalisation** — rewrites the SQL to canonical SQLite syntax via `stmt.sql(dialect="sqlite")`.

If sqlglot is not installed the node passes transparently (`sqlglot_valid=True`).

### Routing

```
valid   → Executor
invalid → Node 3 (SQLWriter, retry with sqlglot error in feedback)
```

---

## Node 5 — ExecutorAgent

**File:** `genai/nl2sql_agent/executor_agent.py`  
**Config:** `genai/config/query.toml`  
**Type:** Rule-based (no LLM call)

### What it does

1. Applies the row-limit policy from `query.toml`:
   - If `enforce_default_limit=true` and the SQL has no `LIMIT` clause, appends `LIMIT <default_limit>`.
2. Opens a `BEGIN DEFERRED` transaction.
3. Runs `cursor.fetchmany(max_display_rows)` — hard ceiling regardless of SQL.
4. **Always rolls back** (unconditional in `finally`).
5. Builds the result summary string.

### Read-only guarantee

```python
conn.execute("BEGIN DEFERRED")   # deferred = no write locks acquired
conn.rollback()                  # always, even on success
```

Even if a malicious SQL somehow bypassed all prior guards, the rollback ensures no data is ever written.

### Outputs written

```
result_rows     — list of {column: value} dicts
result_columns  — ordered list of column names
result_summary  — one-sentence human-readable summary
error           — None on success, error message on failure
stages          — ["executor_agent:ok rows=42"]
```

---

## Adding a New Node

1. Create `genai/nl2sql_agent/my_node.py` with a class that has `.run(state: AgentState) -> AgentState`.
2. Add a `my_node.yaml` under `genai/prompt/` (optional, only if it has user-facing messages).
3. In `genai/nl2sql_agent/graph.py`:
   - Import your node class.
   - Add `graph.add_node("my_node", MyNode().run)`.
   - Wire it with `graph.add_edge(...)` or `graph.add_conditional_edges(...)`.

## Design Decisions

| Decision | Rationale |
|----------|-----------|
| Streamlit replaces FastAPI + run_ui.py | Single process, single port, no proxy layer — simpler to deploy |
| `llm_config.json` for credential persistence | Survives Streamlit hot-reloads and server restarts without `.env` changes |
| All config in TOML + `.env`, zero hard-coded strings | Environment-specific values never touch source code |
| YAML prompts loaded via `utils/registry.py` | Change LLM wording without touching Python; hot-reload via `reload()` |
| `total=False` TypedDict for AgentState | Each node writes only its own keys; LangGraph merges correctly |
| sqlglot AST validation **after** regex review | Defence in depth — catches syntax errors the regex misses |
| `BEGIN DEFERRED` + unconditional rollback in executor | SQLite stays read-only regardless of what SQL reaches the executor |
| Logs always anchored to `<project_root>/data/logs/` | Consistent location regardless of which directory the server starts from |

## `llm_config.json` — Runtime Credential Persistence

When you click **Test & Save** on the Streamlit LLM Configuration page, credentials are written to `llm_config.json` in the project root:

```json
{
    "model_name": "google/gemma-4-31b-it",
    "base_url": "http://10.190.236.15:9000/api/v1",
    "api_key": "EMPTY"
}
```

On next startup `app.py` reads this file and pre-fills the form inputs. This means you do not need to re-enter credentials after a Streamlit restart.

> `llm_config.json` is created in the **project root** (one level above `dlh_ai/`).  
> Add it to `.gitignore` if the key is sensitive.

# Guardrails

The pipeline has **four independent guard layers**.  None of them call the LLM — they are all rule-based and therefore fast, deterministic, and zero-cost.

```
User question
      │
      ▼
 ┌─────────────────────────────┐
 │  Layer 1: IntentChecker     │  Blocks empty / harmful / vague input
 └──────────────┬──────────────┘
                │ passes
                ▼
 ┌─────────────────────────────┐
 │  LLM generates SQL          │
 └──────────────┬──────────────┘
                │
                ▼
 ┌─────────────────────────────┐
 │  Layer 2: SQLReviewer       │  Blocks DML, checks structure + tables
 └──────────────┬──────────────┘
                │ passes
                ▼
 ┌─────────────────────────────┐
 │  Layer 3: SQLValidator      │  AST parse, SELECT-only gate
 └──────────────┬──────────────┘
                │ passes
                ▼
 ┌─────────────────────────────┐
 │  Layer 4: ExecutorAgent     │  BEGIN DEFERRED + unconditional ROLLBACK
 └──────────────┬──────────────┘
                │
                ▼
          Result rows
```

---

## Layer 1 — IntentChecker

**File:** `genai/nl2sql_agent/intent_checker.py`  
**When:** Before anything reaches the LLM.

### Blocked inputs

| Type | Examples | Response |
|------|----------|----------|
| Empty question | `""`, `"   "` | "No question was provided…" |
| Disallowed action word | `"delete accounts"`, `"drop table"` | "⚠ Sorry, the keyword DELETE is not permitted…" |
| Too short / meaningless | `"pending"`, `"what is"` | "Your question is too short or unclear…" |
| Vague / ambiguous | `"give me data"`, `"please verify this"` | "Your question is unclear. Please rephrase…" |

### Disallowed action keywords

```
delete  drop  insert  update  truncate  alter  create  replace
pragma  attach  detach  vacuum  modify  remove  erase  wipe  purge  destroy
```

### Customising messages

All user-facing strings live in `genai/prompt/intent_checker.yaml`.  Edit that file to change wording — no Python change required.

```yaml
messages:
  empty_question:      "No question was provided…"
  disallowed_action:   "⚠ Sorry, the keyword {blocked_word} is not permitted…"
  too_short:           "Your question is too short or unclear to process."
  clarification:       "Your question is unclear. Please rephrase with more detail."
```

### Adding domain metric terms

Edit `_METRIC_TERMS` in `intent_checker.py` to add domain-specific terms that should bypass the too-short check:

```python
_METRIC_TERMS: Final[frozenset[str]] = frozenset({
    "age", "kyc", "balance", "maturity", "product", "branch", "circle",
    # ← add your terms here, lowercase
})
```

---

## Layer 2 — SQLReviewer

**File:** `genai/nl2sql_agent/sql_reviewer.py`  
**When:** After the LLM generates SQL.

### Checks

| Check | Trigger | Action |
|-------|---------|--------|
| Retry ceiling | `retry_count >= max_retries` (default 3) | Sets `error` field → Executor surfaces the message to user |
| Empty SQL | `sql_query == ""` | `retry_count += 1` → back to SQLWriter |
| Disallowed keyword | Any of INSERT/UPDATE/DELETE/DROP/TRUNCATE/ALTER/CREATE/REPLACE/PRAGMA/ATTACH/DETACH/VACUUM | `retry_count += 1` → back to SQLWriter |
| No SELECT prefix | SQL doesn't start with `SELECT` | `retry_count += 1` → back to SQLWriter |
| Missing FROM | No `FROM` keyword found | `retry_count += 1` → back to SQLWriter |
| Unknown table | Any table in `FROM`/`JOIN` clauses doesn't exist in the `.db` | `retry_count += 1` → back to SQLWriter |

### Changing max retries

In `genai/nl2sql_agent/graph.py`:

```python
compiled_graph = build_nl2sql_graph(
    model_service=model_service,
    db_path=db_path,
    max_review_retries=5,   # ← change here (default: 3)
)
```

### Customising feedback strings

All reviewer feedback strings live in `genai/prompt/sql_review.yaml`:

```yaml
feedback:
  empty_sql:        "The SQL Writer returned an empty query."
  no_select:        "The query must begin with SELECT."
  missing_from:     "The query is missing a FROM clause."
  disallowed_kw:    "Keyword {kw} is not allowed in compliance queries."
  unknown_tables:   "The following tables do not exist: {tables}."
  max_retries:      "⚠ The request could not be completed after {max_retries} attempts."
```

---

## Layer 3 — SQLValidator

**File:** `genai/nl2sql_agent/sql_validator.py`  
**When:** After SQLReviewer passes.

### Checks

| Check | What it catches | Example |
|-------|----------------|---------|
| **Parse validity** | Syntax errors the regex reviewer missed | `SELECT * FORM tbl` (typo) |
| **Statement type** | Non-SELECT statements that regex missed | `EXPLAIN SELECT …` (type = `Command`) |
| **Dialect normalisation** | Vendor-specific syntax not valid in SQLite | Rewrites to canonical SQLite |

### Changing dialect

In `genai/nl2sql_agent/graph.py`:

```python
compiled_graph = build_nl2sql_graph(
    ...
    sqlglot_dialect="sqlite",   # ← change here if you use a different DB engine
)
```

### sqlglot not installed

If sqlglot is not installed the node automatically skips and sets `sqlglot_valid=True`.

---

## Layer 4 — ExecutorAgent

**File:** `genai/nl2sql_agent/executor_agent.py`  
**When:** After all prior layers pass.

### Read-only enforcement

```python
conn.execute("BEGIN DEFERRED")   # No write locks acquired
cursor = conn.execute(sql)       # SELECT only reaches here
# ...
conn.rollback()                  # Always — even if execute() succeeded
conn.close()
```

`BEGIN DEFERRED` means SQLite acquires no lock at all until the first read.  Because we always rollback, **no write can ever be committed** regardless of what SQL text reaches the executor.

### Row-limit policy

Controlled entirely by `genai/config/query.toml`:

```toml
[query]
enforce_default_limit = false   # true = auto-append LIMIT if SQL has none
default_limit         = 10      # rows appended when enforce=true
max_display_rows      = 500     # hard ceiling via fetchmany()
```

`max_display_rows` applies regardless of `enforce_default_limit` — it is a hard memory/performance ceiling enforced by `cursor.fetchmany(max_display_rows)`.

---

## Defence-in-Depth Matrix

| Attack / mistake | Layer 1 | Layer 2 | Layer 3 | Layer 4 |
|-----------------|---------|---------|---------|---------|
| `DELETE FROM …` | ✓ (action word) | ✓ (keyword check) | ✓ (non-SELECT AST) | ✓ (rollback) |
| `SELECT 1; DROP TABLE …` | — | ✓ (DROP keyword) | ✓ (multiple stmts) | ✓ (rollback) |
| SQL with typo | — | — | ✓ (ParseError) | — |
| SQL with wrong table | — | ✓ (table check) | — | — |
| Vague non-query | ✓ (vague check) | — | — | — |
| 10 MB result set | — | — | — | ✓ (fetchmany cap) |

# Logging

## Overview

All logging is bootstrapped through a single factory function:

```python
from utils.logger import get_logger
LOGGER = get_logger("compliance_bot.my_module")
```

**First call** configures the entire Python logging system from `genai/config/logging.toml`.  
Subsequent calls just return the named logger from the standard library cache.

---

## Log Destinations

| Handler | Level | Where | Format |
|---------|-------|-------|--------|
| `console` | INFO | stdout | `asctime \| LEVEL \| name \| module:lineno \| message` |
| `rotating_file` | DEBUG | `data/logs/compliance_bot.log` | same |

The file handler uses `RotatingFileHandler` — it rolls over at **10 MB** and keeps **10 backups**.

---

## Log File Location

The log directory is always `<project_root>/dlh_ai/data/logs/` regardless of which directory you run the app from.

This is enforced in `utils/logger.py`:

```python
_PROJECT_ROOT: Path = Path(__file__).resolve().parents[1]
# ...
resolved_log_dir = raw_dir if raw_dir.is_absolute() else _PROJECT_ROOT / raw_dir
```

So even if Streamlit is started from a different working directory, the log file always ends up at:

```
dlh_ai/data/logs/compliance_bot.log
```

---

## TOML Configuration

`genai/config/logging.toml` — edit this to change log behaviour.

```toml
[logging]
version                  = 1
disable_existing_loggers = false

[logging.formatters.standard]
format  = "%(asctime)s | %(levelname)-8s | %(name)s | %(module)s:%(lineno)d | %(message)s"
datefmt = "%Y-%m-%d %H:%M:%S"

[logging.handlers.console]
class     = "logging.StreamHandler"
level     = "INFO"       # ← change to DEBUG to see all messages in terminal
formatter = "standard"
stream    = "ext://sys.stdout"

[logging.handlers.rotating_file]
class       = "logging.handlers.RotatingFileHandler"
level       = "DEBUG"    # ← file always gets DEBUG and above
formatter   = "standard"
filename    = "data/logs/compliance_bot.log"  # overwritten at runtime with absolute path
maxBytes    = 10485760   # 10 MB
backupCount = 10
encoding    = "utf-8"
delay       = true       # don't create the file until first write

[logging.loggers.compliance_bot]
level     = "DEBUG"
handlers  = ["console", "rotating_file"]
propagate = false        # ← don't bubble to the root logger

[logging.root]
level    = "WARNING"
handlers = ["console"]
```

---

## Enabling Debug Logging

**Option 1 — `.env`:**
```dotenv

```

**Option 2 — TOML:**
```toml

```

## Adding Logging to New Code

```python
from utils.logger import get_logger

LOGGER = get_logger("compliance_bot.my_new_module")
```

# Testing

> The test suite has been **removed** from this version of the project.  
> The codebase is production-ready for the Streamlit deployment.  
> All pipeline logic (agent nodes, registry, helpers) is covered by inline type annotations, docstrings, and the Pydantic schema validation that runs at request time.

If you wish to add tests in future, here is how the test layer was structured.

## Recommended Test Setup

Add pytest and mock dependencies (not in `requirements.txt` for the Streamlit build):

```bash
pip install pytest pytest-mock pytest-cov
```

Create a `dlh_ai/tests/` folder with a `conftest.py`:

```python
import sqlite3
from pathlib import Path
import pytest

@pytest.fixture(scope="session")
def tmp_db(tmp_path_factory) -> Path:
    """Temporary SQLite DB with 5 seeded COMP_TBL rows."""
    db_path = tmp_path_factory.mktemp("db") / "test.db"
    conn = sqlite3.connect(str(db_path))
    conn.execute("""
        CREATE TABLE COMP_TBL (
            BRCD TEXT, CUST_NBR TEXT, AGE_YRS INTEGER,
            KYC_FLG TEXT, CIRCLENAME TEXT, SR_CTZN_ACCT_FLG INTEGER
        )
    """)
    conn.executemany(
        "INSERT INTO COMP_TBL VALUES (?,?,?,?,?,?)",
        [("001","C1",65,"Y","DELHI",1),
         ("002","C2",45,"N","MUMBAI METRO",0),
         ("003","C3",72,"Y","MUMBAI METRO",1),
         ("004","C4",33,"N","CHENNAI",0),
         ("005","C5",61,"Y","DELHI",1)],
    )
    conn.commit(); conn.close()
    return db_path
```

## Example Test Files

| File | Module tested |

"""

class ComplianceBotError(Exception):
    """Base exception for all Compliance Chatbot domain errors."""

    def __init__(self, message: str, detail: str | None = None) -> None:
        super().__init__(message)
        self.message: str = message
        self.detail: str | None = detail

    def __str__(self) -> str:
        return self.message



class ModelLoadError(ComplianceBotError):
    """Raised when an LLM or tokenizer fails to load."""


class ModelInferenceError(ComplianceBotError):
    """Raised when the model produces an unexpected output during inference."""


class LLMProviderError(ComplianceBotError):
    """Raised when an LLM provider call fails (auth, HTTP error, bad response …)."""


class LLMTimeoutError(ComplianceBotError):
    """Raised when an LLM provider call exceeds the configured timeout."""


class UnknownProviderError(ComplianceBotError):
    """Raised when an unrecognised LLM provider name is requested."""



class SqlGenerationError(ComplianceBotError):
    """Raised when the NL-to-SQL pipeline cannot produce a valid SQL statement."""


class SqlExecutionError(ComplianceBotError):
    """Raised when a generated SQL query fails to execute against the database."""



class DatabaseConnectionError(ComplianceBotError):
    """Raised when the application cannot connect to the database."""


class RecordNotFoundError(ComplianceBotError):
    """Raised when an expected database record is absent."""



class DataIngestionError(ComplianceBotError):
    """Raised when raw data cannot be ingested or structured correctly."""


class SchemaValidationError(ComplianceBotError):
    """Raised when an incoming payload fails schema validation."""

"""
genai/core/model.py — LLM provider lifecycle, inference, and management.


All LLM-related code lives here; nothing else in genai/core/ touches the
OpenAI client directly.
"""

import re
import time

import httpx
from openai import APIError, APITimeoutError, OpenAI

from genai.core.exceptions import (
    ModelInferenceError,
    ModelLoadError,
    UnknownProviderError,
)
from utils.config import (
    DEFAULT_MAX_NEW_TOKENS,
    DEFAULT_TEMPERATURE,
    SQL_BEGIN_TAG,
    SQL_END_TAG,
    ModelSettings,
    get_settings,
)
from utils.logger import get_logger

LOGGER = get_logger("compliance_bot.core.model")


class ModelLoaderService:
    """Manages the lifecycle of the configured remote LLM provider."""

    def __init__(self, settings: ModelSettings) -> None:
        self.settings: ModelSettings = settings
        self._client: OpenAI | None = None
        self._send_sampling_params = True

    def load(self) -> None:
        """Load the configured provider resources."""
        self._load_dlh_client()

    def _load_dlh_client(self) -> None:
        """Initialise the DLH OpenAI-compatible client."""
        http_client = None
        if not self.settings.dlh_verify_ssl:
            http_client = httpx.Client(verify=False)  # noqa: S501
        try:
            self._client = OpenAI(
                api_key=self.settings.dlh_api_key,
                base_url=self.settings.dlh_base_url,
                timeout=30.0,
                **({"http_client": http_client} if http_client is not None else {}),
            )
            LOGGER.info("DLH client initialised successfully.")
        except Exception as exc:  # pragma: no cover
            raise ModelLoadError(f"Failed to initialise DLH client: {exc}", detail=str(exc)) from exc

    def is_loaded(self) -> bool:
        """Return True when the configured provider is ready for inference."""
        return self._client is not None

    def test_connection(self) -> None:
        """Verify the configured remote provider connection with a minimal request."""
        if self._client is None:
            raise ModelLoadError("Remote client is not initialised.")

        request_kwargs = {
            "model": self.settings.model_name,
            "messages": [{"role": "user", "content": "Reply with OK."}],
            "temperature": 0.1,
            "max_tokens":  8,
        }

        try:
            self._client.chat.completions.create(**request_kwargs)
        except APITimeoutError as exc:
            raise ModelInferenceError(f"Connection test timed out: {exc}", detail=str(exc)) from exc
        except APIError as exc:
            raise ModelInferenceError(f"Connection test failed: {exc}", detail=str(exc)) from exc

    def generate(
        self,
        prompt: str,
        max_new_tokens: int = DEFAULT_MAX_NEW_TOKENS,
        temperature: float = DEFAULT_TEMPERATURE,
    ) -> str:
        """Run inference on the configured provider."""
        if not self.is_loaded():
            raise ModelInferenceError(
                "Model is not loaded. Call ModelLoaderService.load() before generating."
            )

        request_kwargs = {
            "model": self.settings.model_name,
            "messages": [{"role": "user", "content": prompt}],
            "temperature": temperature,
            "max_tokens":  max_new_tokens,
        }

        try:
            response = self._client.chat.completions.create(**request_kwargs)
            return response.choices[0].message.content or ""
        except APITimeoutError as exc:
            raise ModelInferenceError(f"Provider request timed out: {exc}", detail=str(exc)) from exc
        except APIError as exc:
            raise ModelInferenceError(f"Provider API error: {exc}", detail=str(exc)) from exc

    def extract_sql(self, model_output: str) -> str | None:
        """Extract the SQL statement from delimited model output."""
        pattern = re.compile(
            rf"{re.escape(SQL_BEGIN_TAG)}\s*(.*?)\s*{re.escape(SQL_END_TAG)}",
            re.DOTALL | re.IGNORECASE,
        )
        match = pattern.search(model_output)
        if match:
            sql = match.group(1).strip()
            LOGGER.debug("Extracted SQL: %s", sql)
            return sql
        LOGGER.warning("SQL delimiter tags not found in model output. Raw: %s", model_output[:300])
        return None

    def unload(self) -> None:
        """Release provider resources from memory."""
        LOGGER.info("Unloading model resources …")
        self._client = None
        LOGGER.info("Model unloaded.")



def create_model_service(settings: "ModelSettings | None" = None) -> ModelLoaderService:
    """
    Build and load a ModelLoaderService, then return it ready for inference.

    Pass a ModelSettings object to override the defaults, or leave as None
    to use whatever is configured in .env / AppSettings.
    """
    resolved = settings or get_settings().model
    LOGGER.info(
        "create_model_service — provider=%s model=%s",
        resolved.provider, resolved.model_name,
    )
    service = ModelLoaderService(settings=resolved)
    service.load()
    return service



def generate_with_retry(
    model_service: ModelLoaderService,
    prompt: str,
    max_new_tokens: int = DEFAULT_MAX_NEW_TOKENS,
    temperature: float = DEFAULT_TEMPERATURE,
    retries: int = 2,
    retry_delay: float = 1.0,
) -> str:
    """
    Call model_service.generate() and retry on failure.

    retries — how many additional attempts after the first (total = retries + 1).
    Raises ModelInferenceError if every attempt fails.
    """
    last_exc: ModelInferenceError | None = None
    for attempt in range(retries + 1):
        try:
            return model_service.generate(
                prompt=prompt,
                max_new_tokens=max_new_tokens,
                temperature=temperature,
            )
        except ModelInferenceError as exc:
            last_exc = exc
            if attempt < retries:
                LOGGER.warning(
                    "generate_with_retry — attempt %d/%d failed: %s. Retrying in %.1fs…",
                    attempt + 1, retries + 1, exc, retry_delay,
                )
                time.sleep(retry_delay)

    raise last_exc  # type: ignore[misc]


def extract_text_between(text: str, start_marker: str, end_marker: str) -> str | None:
    """
    Pull the content between start_marker and end_marker out of text.

    Case-insensitive.  Returns None if either marker is missing.
    """
    lower = text.lower()
    start_idx = lower.find(start_marker.lower())
    if start_idx == -1:
        return None
    content_start = start_idx + len(start_marker)
    end_idx = lower.find(end_marker.lower(), content_start)
    if end_idx == -1:
        return None
    return text[content_start:end_idx].strip()


def truncate_prompt(prompt: str, max_chars: int = 8000) -> str:
    """
    Keep the prompt under max_chars by removing characters from the middle.

    Preserving the header (instructions) and the tail (question) gives the
    LLM the most important context even when the schema block is very long.
    """
    if len(prompt) <= max_chars:
        return prompt
    half = max_chars // 2
    LOGGER.warning("truncate_prompt — length %d > %d; cutting the middle.", len(prompt), max_chars)
    return prompt[:half] + "\n...[truncated]...\n" + prompt[-half:]



class ProviderManager:
    """
    Registry and lifecycle manager for multiple LLM provider services.

    Register providers first, then call activate() to make one of them live.
    The active provider is the one all inference calls go through.
    """

    def __init__(self) -> None:
        self._registry: dict[str, ModelSettings] = {}
        self._services: dict[str, ModelLoaderService] = {}
        self._active_name: str | None = None

    def register(self, name: str, settings: ModelSettings) -> None:
        """Store a provider configuration under name (not loaded until activate())."""
        self._registry[name] = settings
        LOGGER.info("ProviderManager — registered provider: %s", name)

    def activate(self, name: str) -> ModelLoaderService:
        """Make name the active provider, loading it if needed."""
        if name not in self._registry:
            raise UnknownProviderError(
                f"Provider '{name}' is not registered. Available: {list(self._registry)}"
            )
        if name not in self._services:
            service = ModelLoaderService(settings=self._registry[name])
            service.load()
            self._services[name] = service
            LOGGER.info("ProviderManager — loaded provider: %s", name)
        self._active_name = name
        LOGGER.info("ProviderManager — active provider: %s", name)
        return self._services[name]

    @property
    def active(self) -> ModelLoaderService:
        """Return the currently active service. Raises RuntimeError if none is set."""
        if self._active_name is None or self._active_name not in self._services:
            raise RuntimeError("No active provider. Call ProviderManager.activate(name) first.")
        return self._services[self._active_name]

    @property
    def active_name(self) -> str | None:
        """Name of the currently active provider, or None."""
        return self._active_name

    def registered_names(self) -> list[str]:
        """Return a sorted list of registered provider names."""
        return sorted(self._registry)

    def unload(self, name: str) -> None:
        """Unload a provider service and free its memory."""
        service = self._services.pop(name, None)
        if service is not None:
            service.unload()
            LOGGER.info("ProviderManager — unloaded provider: %s", name)
        if self._active_name == name:
            self._active_name = None

    def unload_all(self) -> None:
        """Unload every registered provider service."""
        for name in list(self._services):
            self.unload(name)

"""
genai/core/processor.py — compliance orchestration and legacy schema reference.

 — integration point between FastAPI and LangGraph
 — legacy Python schema dict (kept as reference)

ComplianceProcessor is the single entry-point for all compliance queries:
it accepts a ComplianceChatRequest, invokes the compiled LangGraph graph,
and maps the resulting AgentState onto the Pydantic response models.
"""

import uuid
from typing import Any

from genai.core.exceptions import ComplianceBotError
from genai.core.schemas import (
    ComplianceChatRequest,
    ComplianceChatResponse,
    ComplianceOrchestrationResponse,
)
from genai.nl2sql_agent.state import AgentState
from utils.logger import get_logger

LOGGER = get_logger("compliance_bot.core.processor")



class ComplianceProcessor:
    """
    Orchestrates the NL2SQL compliance query pipeline via LangGraph.

    Intentionally thin — all multi-agent logic lives inside the agent nodes
    and graph wiring.  This class maps between API models and AgentState.
    """

    def __init__(self, compiled_graph: Any) -> None:
        self._graph = compiled_graph

    def process_query(self, request: ComplianceChatRequest) -> ComplianceChatResponse:
        """Run the agentic pipeline and return the simplified chat response."""
        orchestration = self.process_query_with_orchestration(request)
        return ComplianceChatResponse(
            question_id=orchestration.question_id,
            sql_query=orchestration.sql_query,
            df_output=orchestration.df_output,
            result_columns=orchestration.result_columns,
            result_summary=orchestration.result_summary,
            error=orchestration.error or orchestration.clarification_question,
        )

    def process_query_with_orchestration(
        self,
        request: ComplianceChatRequest,
    ) -> ComplianceOrchestrationResponse:
        """Run the full LangGraph NL2SQL pipeline and return the detailed response."""
        question_id = request.question_id or str(uuid.uuid4())
        question = (request.question or "").strip()

        if not question:
            LOGGER.warning("[%s] Empty question received.", question_id)
            return ComplianceOrchestrationResponse(
                question_id=question_id,
                error="No question was provided in the request.",
                stages=["validation:empty_question"],
            )

        LOGGER.info("[%s] Invoking NL2SQL LangGraph for: %r", question_id, question)

        initial_state: AgentState = AgentState(  # type: ignore[misc]
            question=question,
            question_id=question_id,
            stages=[],
        )

        try:
            final_state: AgentState = self._graph.invoke(initial_state)
        except ComplianceBotError:
            raise
        except Exception:  # pylint: disable=broad-exception-caught

            error=error,
        )

"""
Executor Agent — Node 5 of the NL2SQL LangGraph pipeline.

Executes the validated SQL SELECT statement against the target SQLite
database, converts the raw cursor output into a list of column→value
dictionaries, and produces a concise human-readable summary.

The executor operates in strict read-only mode by opening every connection
inside a ``BEGIN DEFERRED`` transaction that is rolled back unconditionally
after execution.

This module also contains result formatting helpers (previously result_formatter.py):
  - format_summary    — one-sentence result summary
  - to_markdown_table — GitHub-Flavoured Markdown table
  - to_csv            — CSV export with header row
  - extract_scalar    — pull single value from aggregate result

Row-limit policy (controlled via genai/config/query.toml)
---------------------------------------------------------
enforce_default_limit = true
    If the SQL has no LIMIT clause, append "LIMIT <default_limit>" before
    executing so unintentionally large result sets are never returned.

enforce_default_limit = false
    Execute the SQL exactly as written — no automatic LIMIT is added.

max_display_rows
    Hard ceiling enforced via fetchmany() regardless of any LIMIT in the SQL.
"""

import contextlib
import csv
import io
import re
import sqlite3
from pathlib import Path
from typing import Any

from genai.nl2sql_agent.state import AgentState
from utils.logger import get_logger

LOGGER = get_logger("compliance_bot.nl2sql_agent.executor_agent")

_LIMIT_RE = re.compile(r"\bLIMIT\b", re.IGNORECASE)

# Config TOML path.
_QUERY_TOML = Path(__file__).resolve().parents[2] / "genai" / "config" / "query.toml"


def _load_query_config() -> dict[str, Any]:
    """Load query.toml — returns defaults on any failure."""
    try:
        try:
            import tomllib
        except ImportError:
            import tomli as tomllib  # type: ignore[no-redef]
        with _QUERY_TOML.open("rb") as fh:
            return tomllib.load(fh).get("query", {})
    except Exception:  # noqa: BLE001
        return {}


_Q: dict[str, Any]     = _load_query_config()
_ENFORCE_LIMIT: bool   = bool(_Q.get("enforce_default_limit", True))
_DEFAULT_LIMIT: int    = int(_Q.get("default_limit", 10))
_MAX_DISPLAY_ROWS: int = int(_Q.get("max_display_rows", 500)) or 500


def _ensure_limit(sql: str) -> tuple[str, bool]:
    """
    Apply the row-limit policy from query.toml.

    Returns *(effective_sql, injected)* where *injected* is True when
    ``LIMIT <default_limit>`` was automatically appended.

    When ``enforce_default_limit = false`` this is a no-op.
    """
    if not _ENFORCE_LIMIT:
        return sql, False
    stripped = sql.rstrip("; \t\n")
    if _LIMIT_RE.search(stripped):
        return sql, False          # SQL already has a LIMIT — leave it alone.
    return f"{stripped} LIMIT {_DEFAULT_LIMIT};", True



def format_summary(
    rows: list[dict[str, object]],
    columns: list[str],
    limit_injected: bool = False,
    default_limit: int = _DEFAULT_LIMIT,
    max_display_rows: int = _MAX_DISPLAY_ROWS,
) -> str:
    """
    Build a one-sentence result summary shown to the user.

    Handles three cases:
    - No rows → tells the user nothing matched.
    - Single-cell aggregate (COUNT, SUM …) → highlights the value.
    - Multi-row result → reports count and column names.
    """
    count = len(rows)

    if count == 0:
        return "The query executed successfully but returned no matching records."

    # Single row, single column — probably an aggregate like COUNT(*).
    if count == 1 and len(rows[0]) == 1:
        col, val = next(iter(rows[0].items()))
        return f"Result: **{col}** = `{val}`"

    if count >= max_display_rows:
        limit_note = f" (first {max_display_rows} shown — add a LIMIT clause to narrow results)"
    elif limit_injected:
        limit_note = (
            f" (showing first {default_limit} rows — "
            f"ask for more with 'show top N' or set enforce_default_limit=false in query.toml)"
        )
    else:
        limit_note = ""

    col_list = ", ".join(columns) if columns else ""
    return f"Query returned **{count} record(s)**{limit_note}. Columns: {col_list}."


def to_markdown_table(
    rows: list[dict[str, object]],
    columns: list[str],
    max_rows: int = 20,
) -> str:
    """
    Render rows as a GitHub-Flavoured Markdown table string.

    Includes at most max_rows data rows; appends a note if rows were trimmed.
    Returns "(no rows)" if the result set is empty.
    """
    if not rows:
        return "(no rows)"

    header    = "| " + " | ".join(columns) + " |"
    separator = "| " + " | ".join("---" for _ in columns) + " |"

    data_rows = []
    for row in rows[:max_rows]:
        cells = [str(row.get(col, "")) for col in columns]
        data_rows.append("| " + " | ".join(cells) + " |")

    lines = [header, separator, *data_rows]
    if len(rows) > max_rows:
        lines.append(f"*… {len(rows) - max_rows} more row(s) not shown.*")

    return "\n".join(lines)


def to_csv(rows: list[dict[str, object]], columns: list[str]) -> str:
    """Render rows as a CSV string with a header row."""
    output = io.StringIO()
    writer = csv.DictWriter(
        output, fieldnames=columns, extrasaction="ignore", lineterminator="\n"
    )
    writer.writeheader()
    writer.writerows(rows)
    return output.getvalue()


def extract_scalar(rows: list[dict[str, object]]) -> object | None:
    """
    Pull the single value from a one-row, one-column result (e.g. COUNT(*)).

    Returns None if the result has more than one row or more than one column.
    """
    if len(rows) == 1 and len(rows[0]) == 1:
        return next(iter(rows[0].values()))
    return None



class ExecutorAgent:
    """
    LangGraph node that executes the approved SQL query and formats results.

    Row-limit behaviour is driven entirely by genai/config/query.toml —
    no code changes needed to adjust limits.
    """

    def run(self, state: AgentState) -> AgentState:
        """Execute the SQL query and return formatted results."""
        sql: str = state.get("sql_query", "").strip()
        db_path_str: str = state.get("db_path", "")

        LOGGER.info("ExecutorAgent — executing SQL: %s", sql)

        if not sql:
            msg = "ExecutorAgent received an empty SQL query."
            LOGGER.error(msg)
            return AgentState(  # type: ignore[return-value]
                result_rows=[],
                result_columns=[],
                result_summary="No query to execute.",
                error=msg,
                stages=["executor_agent:empty_sql"],
            )

        if not db_path_str:
            msg = "ExecutorAgent: database path is not set in the pipeline state."
            LOGGER.error(msg)
            return AgentState(  # type: ignore[return-value]
                result_rows=[],
                result_columns=[],
                result_summary="Database path missing.",
                error=msg,
                stages=["executor_agent:no_db_path"],
            )

        db_path = Path(db_path_str)
        if not db_path.exists():
            msg = f"ExecutorAgent: database file not found at {db_path}."
            LOGGER.error(msg)
            return AgentState(  # type: ignore[return-value]
                result_rows=[],
                result_columns=[],
                result_summary="Database file not found.",
                error=msg,
                stages=["executor_agent:db_not_found"],
            )

        rows, columns, error_msg, limit_injected = self._execute_read_only(sql, db_path)

        if error_msg:
            LOGGER.error("ExecutorAgent — SQL execution error: %s", error_msg)
            return AgentState(  # type: ignore[return-value]
                result_rows=[],
                result_columns=[],
                result_summary="Query execution failed.",
                error=error_msg,
                stages=["executor_agent:error"],
            )

        summary = format_summary(rows, columns, limit_injected)
        LOGGER.info("ExecutorAgent — %d rows returned.", len(rows))

        return AgentState(  # type: ignore[return-value]
            result_rows=rows,
            result_columns=columns,
            result_summary=summary,
            error=None,
            stages=[f"executor_agent:ok rows={len(rows)}"],
        )

    @staticmethod
    def _execute_read_only(
        sql: str,
        db_path: Path,
    ) -> tuple[list[dict[str, object]], list[str], str | None, bool]:
        """
        Execute sql inside a read-only deferred transaction.

        Returns ``(rows, column_names, error_message, limit_injected)``.
        """
        effective_sql, limit_injected = _ensure_limit(sql)
        if limit_injected:
            LOGGER.info(
                "ExecutorAgent — no LIMIT in SQL; auto-applied LIMIT %d (query.toml).",
                _DEFAULT_LIMIT,
            )

        conn: sqlite3.Connection | None = None
        try:
            conn = sqlite3.connect(str(db_path), check_same_thread=False)
            conn.row_factory = sqlite3.Row
            conn.execute("BEGIN DEFERRED")

            cursor = conn.execute(effective_sql)
            col_names: list[str] = [d[0] for d in (cursor.description or [])]
            raw_rows = cursor.fetchmany(_MAX_DISPLAY_ROWS)
            rows: list[dict[str, object]] = [dict(zip(col_names, row, strict=False)) for row in raw_rows]
            return rows, col_names, None, limit_injected

        except sqlite3.OperationalError as op_err:
            return [], [], f"SQL operational error: {op_err} | Query: {effective_sql}", False
        except sqlite3.DatabaseError as db_err:
            return [], [], f"SQLite database error: {db_err} | Query: {effective_sql}", False
        finally:
            if conn is not None:
                with contextlib.suppress(sqlite3.Error):
                    conn.rollback()
                conn.close()

"""
genai/nl2sql_agent/graph.py — LangGraph NL2SQL StateGraph factory.

Wires the six pipeline nodes and exposes:
  - build_nl2sql_graph — full constructor, all parameters explicit
  - create_graph       — convenience wrapper using AppSettings defaults

Node naming matches the file names:
  intent_checker   → IntentChecker   (intent_checker.py)
  schema_checker   → SchemaChecker   (schema_checker.py)
  sql_writer_agent → SQLWriterAgent  (sql_writer_agent.py)
  sql_reviewer     → SQLReviewer     (sql_reviewer.py)
  sql_validator    → SQLValidator    (sql_validator.py)
  executor_agent   → ExecutorAgent   (executor_agent.py)

Pipeline topology
-----------------

    [START]
       │
       ▼
  intent_checker ──(intent_ok=False)──► clarify_node ──► [END]
       │ (intent_ok=True)
       ▼

       │
       ▼
  sql_writer_agent ◄──────────────────────────────────────────┐
       │                                                       │ retry
       ▼                                                       │
  sql_reviewer ──(passed)──► sql_validator ───────────────────┤
       │                          │                            │
       │ (max retries / error)    │ (valid)    (invalid → retry)
       ▼                          ▼
  executor_agent ◄─────────── executor_agent ──► [END]
"""

from __future__ import annotations

from pathlib import Path
from typing import TYPE_CHECKING, Literal

from langgraph.graph import END, START, StateGraph

if TYPE_CHECKING:
    from langgraph.graph.state import CompiledStateGraph

from genai.core.model import ModelLoaderService
from genai.nl2sql_agent.executor_agent import ExecutorAgent
from genai.nl2sql_agent.intent_checker import IntentChecker
from genai.nl2sql_agent.schema_checker import SchemaChecker
from genai.nl2sql_agent.sql_reviewer import SQLReviewer
from genai.nl2sql_agent.sql_validator import SQLValidator

from utils.config import get_settings
from utils.logger import get_logger

LOGGER = get_logger("compliance_bot.nl2sql_agent.graph")

NODE_INTENT:    str = "intent_checker"
NODE_SCHEMA:    str = "schema_checker"
NODE_WRITER:    str = "sql_writer_agent"
NODE_REVIEWER:  str = "sql_reviewer"
NODE_VALIDATOR: str = "sql_validator"
NODE_EXECUTOR:  str = "executor_agent"
NODE_CLARIFY:   str = "clarify_node"


def _route_after_intent(state: AgentState) -> Literal["schema_checker", "clarify_node"]:
    """Clear intent → schema; unclear → clarify."""
    return NODE_SCHEMA if state.get("intent_ok", False) else NODE_CLARIFY


def _route_after_reviewer(
    state: AgentState,
) -> Literal["sql_validator", "sql_writer_agent", "executor_agent"]:
    """After review: hard error → executor (surfaces error), passed → validator, failed → writer retry."""
    if state.get("error"):
        return NODE_EXECUTOR
    if state.get("sql_review_passed", False):
        return NODE_VALIDATOR
    return NODE_WRITER


def _route_after_validator(
    state: AgentState,
) -> Literal["executor_agent", "sql_writer_agent"]:
    """AST valid → executor; invalid → retry writer."""
    return NODE_EXECUTOR if state.get("sqlglot_valid", False) else NODE_WRITER


def _clarify_node(state: AgentState) -> AgentState:
    """Terminal node — routes here when intent is too ambiguous."""
    LOGGER.info("ClarifyNode — ending graph (clarification requested).")
    return AgentState(stages=["clarify_node:terminal"])  # type: ignore[return-value]


def build_nl2sql_graph(
    model_service: ModelLoaderService,
    db_path: Path,
    max_review_retries: int = 3,
    sqlglot_dialect: str = "sqlite",
) -> CompiledStateGraph:  # type: ignore[name-defined]
    """
    Construct and compile the six-node NL2SQL StateGraph.

    model_service      — loaded ModelLoaderService for LLM inference
    db_path            — absolute path to the primary SQLite database file
    max_review_retries — maximum writer → reviewer retry cycles
    sqlglot_dialect    — sqlglot dialect (default: "sqlite")

    Returns a compiled LangGraph StateGraph ready to invoke or stream.
    """
    LOGGER.info(
        "Building NL2SQL LangGraph — db=%s retries=%d dialect=%s",
        db_path.name, max_review_retries, sqlglot_dialect,
    )

    intent_checker = IntentChecker()
    schema_checker = SchemaChecker(data_root=db_path.parent, default_db_path=db_path)
    sql_writer     = SQLWriterAgent(model_service=model_service)
    sql_reviewer   = SQLReviewer(max_retries=max_review_retries)
    sql_validator  = SQLValidator(dialect=sqlglot_dialect)
    executor       = ExecutorAgent()

    graph = StateGraph(AgentState)

    graph.add_node(NODE_INTENT,    intent_checker.run)
    graph.add_node(NODE_SCHEMA,    schema_checker.run)
    graph.add_node(NODE_WRITER,    sql_writer.run)
    graph.add_node(NODE_REVIEWER,  sql_reviewer.run)
    graph.add_node(NODE_VALIDATOR, sql_validator.run)
    graph.add_node(NODE_EXECUTOR,  executor.run)
    graph.add_node(NODE_CLARIFY,   _clarify_node)

    graph.add_edge(START, NODE_INTENT)

    graph.add_conditional_edges(
        NODE_INTENT,
        _route_after_intent,
        {NODE_SCHEMA: NODE_SCHEMA, NODE_CLARIFY: NODE_CLARIFY},
    )

    graph.add_edge(NODE_SCHEMA,  NODE_WRITER)
    graph.add_edge(NODE_WRITER,  NODE_REVIEWER)

    graph.add_conditional_edges(
        NODE_REVIEWER,
        _route_after_reviewer,
        {NODE_VALIDATOR: NODE_VALIDATOR, NODE_EXECUTOR: NODE_EXECUTOR, NODE_WRITER: NODE_WRITER},
    )

    graph.add_conditional_edges(
        NODE_VALIDATOR,
        _route_after_validator,
        {NODE_EXECUTOR: NODE_EXECUTOR, NODE_WRITER: NODE_WRITER},
    )

    graph.add_edge(NODE_EXECUTOR, END)
    graph.add_edge(NODE_CLARIFY,  END)

    compiled = graph.compile()
    LOGGER.info("NL2SQL LangGraph compiled successfully.")
    return compiled


def create_graph(
    model_service: ModelLoaderService,
    db_path: Path | None = None,
    max_review_retries: int = 3,
    sqlglot_dialect: str = "sqlite",
) -> CompiledStateGraph:  # type: ignore[name-defined]
    """
    Convenience wrapper — db_path defaults to the database in AppSettings.

    Returns a compiled LangGraph StateGraph ready to invoke or stream.
    """
    resolved: Path = db_path or get_settings().database.db_path
    LOGGER.info("create_graph — db=%s retries=%d", resolved.name, max_review_retries)
    return build_nl2sql_graph(
        model_service=model_service,
        db_path=resolved,
        max_review_retries=max_review_retries,
        sqlglot_dialect=sqlglot_dialect,
    )

"""
genai/nl2sql_agent/intent_checker.py — Node 1 of the NL2SQL pipeline.

Classifies whether the user question has clear enough intent to proceed.
All user-facing messages are loaded from genai/prompt/intent_checker.yaml —
no message strings are hardcoded in this file.

Four checks (rule-based, no LLM):
  1. Empty question
  2. Disallowed action words (DELETE, DROP, INSERT …)
  3. Too short / meaningless input
  4. Vague / ambiguous phrasing
"""

import re
from typing import Final

from genai.nl2sql_agent.state import AgentState
from utils.logger import get_logger
from utils.registry import get_prompt

LOGGER = get_logger("compliance_bot.nl2sql_agent.intent_checker")

_YAML = "intent_checker.yaml"

_METRIC_TERMS: Final[frozenset[str]] = frozenset(
    {
        "age", "kyc", "balance", "maturity", "product", "branch", "circle",
        "senior", "account", "rate", "deposit", "customer", "flag", "date",
        "interest", "tenor", "brcd", "count", "sum", "total", "list", "show",
        "top", "max", "min", "average",
    }
)

_VAGUE_PHRASES: Final[frozenset[str]] = frozenset(
    {
        "tell me something", "give me data", "get data", "details", "info",
        "information", "data", "give me", "tell me",
    }
)

_AMBIGUOUS_TOKENS: Final[frozenset[str]] = frozenset(
    {"this", "that", "those", "these", "it", "something"}
)

_MIN_MEANINGFUL_TOKENS: Final[int] = 3
_MIN_TOKEN_LEN: Final[int] = 2
_RICH_QUESTION_THRESHOLD: Final[int] = 3

_STOP_WORDS: Final[frozenset[str]] = frozenset(
    {
        "a", "an", "the", "is", "are", "was", "were", "be", "been", "being",
        "have", "has", "had", "do", "does", "did", "will", "would", "could",
        "should", "may", "might", "shall", "can", "need", "dare", "ought",
        "used", "to", "of", "in", "on", "at", "by", "for", "with", "about",
        "against", "between", "through", "during", "before", "after", "above",
        "below", "from", "up", "down", "into", "out", "off", "over", "under",
        "again", "then", "once", "and", "but", "or", "nor", "so", "yet",
        "both", "either", "neither", "not", "no", "only", "own", "same",
        "than", "too", "very", "just", "because", "as", "until", "while",
        "i", "me", "my", "we", "our", "you", "your", "he", "she", "they",
        "him", "her", "his", "its", "their", "what", "which", "who", "whom",
        "how", "when", "where", "why", "all", "any", "each", "every", "some",
        "few", "more", "most", "other", "such", "get", "give", "tell", "let",
        "please", "hi", "hello", "hey", "ok", "okay", "thanks", "thank",
    }
)

_PUNCT_STRIP: re.Pattern[str] = re.compile(r"[^\w]")

_DISALLOWED_ACTION_WORDS: Final[frozenset[str]] = frozenset(
    {
        "delete", "drop", "insert", "update", "truncate", "alter", "create",
        "replace", "pragma", "attach", "detach", "vacuum", "modify", "remove",
        "erase", "wipe", "purge", "destroy",
    }
)


class IntentChecker:
    """
    LangGraph node that classifies user intent.

    All messages sourced from intent_checker.yaml — edit that file to change
    the wording shown to users without touching Python code.
    """

    def run(self, state: AgentState) -> AgentState:
        """Classify intent through four sequential rule-based checks."""
        question: str = state.get("question", "").strip()
        LOGGER.info("IntentChecker — analysing: %r", question)

        # ── Check 1: empty question ────────────────────────────────────────
        if not question:
            LOGGER.warning("IntentChecker — empty question.")
            return AgentState(  # type: ignore[return-value]
                interpreted_question="",
                intent_ok=False,
                clarification_needed=get_prompt(
                    _YAML, "messages", "empty_question",
                    default=(
                        "No question was provided. Please ask a compliance query, "
                        "for example: \"How many accounts have KYC flag N?\""
                    ),
                ),
                stages=["intent_checker:empty"],
            )

        lower_q     = question.lower()
        raw_tokens  = lower_q.split()
        clean_tokens = {_PUNCT_STRIP.sub("", t) for t in raw_tokens if _PUNCT_STRIP.sub("", t)}

        # ── Check 2: disallowed action words ──────────────────────────────
        matched = _DISALLOWED_ACTION_WORDS & clean_tokens
        if matched:
            blocked_word = min(matched).upper()
            LOGGER.warning("IntentChecker — disallowed action word: %s", blocked_word)
            template = get_prompt(_YAML, "messages", "disallowed_action", default="")
            msg = template.format(blocked_word=blocked_word) if template else (
                f"⚠ Sorry, the keyword **{blocked_word}** is not permitted. "
                "This system is read-only — only SELECT queries are supported."
            )
            return AgentState(  # type: ignore[return-value]
                interpreted_question=self._normalise(question),
                intent_ok=False,
                clarification_needed=msg,
                stages=[f"intent_checker:disallowed_action_word={blocked_word}"],
            )

        # ── Check 3: too short / meaningless ──────────────────────────────
        has_metric = bool(_METRIC_TERMS & clean_tokens) or any(m in lower_q for m in _METRIC_TERMS)
        meaningful = {
            t for t in clean_tokens
            if t not in _STOP_WORDS and len(t) >= _MIN_TOKEN_LEN and not t.isdigit()
        }
        if len(meaningful) < _MIN_MEANINGFUL_TOKENS and not has_metric:
            LOGGER.info("IntentChecker — too short (%d tokens): %r", len(meaningful), question)
            return AgentState(  # type: ignore[return-value]
                interpreted_question=self._normalise(question),
                intent_ok=False,
                clarification_needed=get_prompt(
                    _YAML, "messages", "too_short",
                    default="Your question is too short or unclear to process.",
                ),
                stages=["intent_checker:too_short"],
            )

        # ── Check 4: vague / ambiguous phrasing ───────────────────────────
        is_pure_vague   = lower_q.strip("?. ") in _VAGUE_PHRASES
        has_ambiguous   = bool(_AMBIGUOUS_TOKENS & clean_tokens)
        rich_question   = len(meaningful) > _RICH_QUESTION_THRESHOLD
        needs_clarification = is_pure_vague or (has_ambiguous and not has_metric and not rich_question)

        interpreted = self._normalise(question)

        if needs_clarification:
            LOGGER.info("IntentChecker — clarification required: %r", question)
            return AgentState(  # type: ignore[return-value]
                interpreted_question=interpreted,
                intent_ok=False,
                clarification_needed=get_prompt(
                    _YAML, "messages", "clarification",
                    default="Your question is unclear. Please rephrase with more detail.",
                ),
                stages=["intent_checker:clarification_required"],
            )

        LOGGER.info("IntentChecker — intent OK: %r", interpreted)
        return AgentState(  # type: ignore[return-value]
            interpreted_question=interpreted,
            intent_ok=True,
            clarification_needed=None,
            stages=["intent_checker:ok"],
        )

    @staticmethod
    def _normalise(question: str) -> str:
        """Strip whitespace and ensure a trailing question mark."""
        q = question.strip()
        if q and not q.endswith("?"):
            q = q.rstrip(".") + "?"
        return q

"""
genai/nl2sql_agent/schema_checker.py — Node 2 of the NL2SQL pipeline.

Introspects the live SQLite database and builds a Markdown-formatted schema
context string for the SQL Writer to embed in its LLM prompt.

Configuration (sample_row_limit, header templates) is loaded from
genai/prompt/schema_checker.yaml.  Column descriptions come from
genai/config/databases.toml via the registry.

Also contains low-level SQLite introspection helpers used only here.
"""

import sqlite3
from pathlib import Path

from genai.nl2sql_agent.state import AgentState
from utils.logger import get_logger
from utils.registry import get_prompt, get_table_meta, load_prompt

LOGGER = get_logger("compliance_bot.nl2sql_agent.schema_checker")

_YAML     = "schema_checker.yaml"
_CFG      = load_prompt(_YAML)
_SAMPLE_N = int(_CFG.get("schema", {}).get("sample_row_limit", 3))

ColumnInfo = tuple[int, str, str, int, str | None, int]


def load_table_names(db_path: str | Path) -> list[str]:
    """Return a sorted list of all table names in the database file."""
    path = Path(db_path)
    if not path.exists():
        return []
    try:
        with sqlite3.connect(str(path)) as conn:
            rows = conn.execute(
                "SELECT name FROM sqlite_master WHERE type='table' ORDER BY name"
            ).fetchall()
        return [row[0] for row in rows]
    except sqlite3.Error:
        LOGGER.exception("schema_checker.load_table_names — %s", path)
        return []


def load_columns(db_path: str | Path, table_name: str) -> list[ColumnInfo]:
    """Return PRAGMA table_info rows for the given table."""
    path = Path(db_path)
    if not path.exists():
        return []
    try:
        with sqlite3.connect(str(path)) as conn:
            return conn.execute(f"PRAGMA table_info('{table_name}')").fetchall()
    except sqlite3.Error:
        LOGGER.exception("schema_checker.load_columns — %s.%s", path, table_name)
        return []


def load_sample_rows(
    db_path: str | Path,
    table_name: str,
    limit: int = 3,
) -> tuple[list[str], list[tuple]]:  # type: ignore[type-arg]
    """Return (column_names, rows) for up to limit sample rows."""
    path = Path(db_path)
    if not path.exists():
        return [], []
    try:
        with sqlite3.connect(str(path)) as conn:
            cursor = conn.execute(f"SELECT * FROM '{table_name}' LIMIT {limit}")
            col_names = [d[0] for d in (cursor.description or [])]
            return col_names, cursor.fetchall()
    except sqlite3.Error:
        LOGGER.exception("schema_checker.load_sample_rows — %s", path)
        return [], []


class SchemaChecker:
    """
    LangGraph node — selects the best matching SQLite database for the question
    and builds a rich Markdown schema context string for the SQL Writer.

    Section headers and error messages are loaded from schema_checker.yaml.
    Column business descriptions are merged from databases.toml (via registry).
    """

    def __init__(self, data_root: Path, default_db_path: Path) -> None:
        self._data_root      = data_root
        self._default_db     = default_db_path

    def run(self, state: AgentState) -> AgentState:
        """Select the best database and build the schema context string."""
        question: str = state.get("interpreted_question", state.get("question", ""))
        LOGGER.info("SchemaChecker — selecting database for: %r", question)

        db_path = self._select_database(question)
        context = self._build_context(db_path)

        LOGGER.info("SchemaChecker — using db=%s", db_path.name)
        return AgentState(  # type: ignore[return-value]
            db_path=str(db_path),
            schema_context=context,
            stages=[f"schema_checker:db={db_path.name}"],
        )

    def _select_database(self, question: str) -> Path:
        """Choose the most relevant .db file for the question."""
        candidates: list[Path] = sorted(self._data_root.glob("**/*.db"))
        if not candidates:
            LOGGER.warning(
                "SchemaChecker — %s",
                get_prompt(_YAML, "messages", "no_db_found", default="No .db files found."),
            )
            return self._default_db

        lower_q    = question.lower()
        best       = self._default_db
        best_score = -1
        for db in candidates:
            score = self._score(db, lower_q)
            if score > best_score:
                best_score, best = score, db
        return best

    @staticmethod
    def _score(db_path: Path, lower_q: str) -> int:
        """Keyword-overlap relevance score for a database against the question."""
        score = 1 if db_path.stem.lower() in lower_q else 0
        try:
            with sqlite3.connect(str(db_path)) as conn:
                for (tbl,) in conn.execute(
                    "SELECT name FROM sqlite_master WHERE type='table'"
                ).fetchall():
                    if tbl.lower() in lower_q:
                        score += 2
                    for col in conn.execute(f"PRAGMA table_info('{tbl}')").fetchall():
                        if col[1].lower() in lower_q:
                            score += 1
        except sqlite3.Error:
            pass
        return score

    def _build_context(self, db_path: Path) -> str:
        """Build the Markdown schema context block for LLM injection."""
        # Header templates from schema_checker.yaml (with fallbacks)
        schema_header   = get_prompt(_YAML, "messages", "schema_header",   default="## Database: `{db_stem}`")
        table_header    = get_prompt(_YAML, "messages", "table_header",    default="### Table: `{table_name}`")
        col_desc_header = get_prompt(
            _YAML, "messages", "col_desc_header",
            default="**Column descriptions for `{table_name}` (from databases.toml):**",
        )
        sample_header   = get_prompt(
            _YAML, "messages", "sample_header",
            default="**Sample rows from `{table_name}`:**",
        )
        schema_error    = get_prompt(_YAML, "messages", "schema_error",    default="*Schema unavailable: {error}*")

        lines: list[str] = [schema_header.format(db_stem=db_path.stem), ""]  # type: ignore[arg-type]

        try:
            with sqlite3.connect(str(db_path)) as conn:
                conn.row_factory = sqlite3.Row
                tables = conn.execute(
                    "SELECT name FROM sqlite_master WHERE type='table' ORDER BY name"
                ).fetchall()

                for (tbl,) in tables:
                    lines.append(table_header.format(table_name=tbl))  # type: ignore[arg-type]
                    lines.append("")
                    lines.append("| # | Column | Type | PK | Not Null |")
                    lines.append("|---|--------|------|----|----------|")

                    cols = conn.execute(f"PRAGMA table_info('{tbl}')").fetchall()
                    for col in cols:
                        cid, name, ctype, notnull, _dflt, pk = col
                        lines.append(
                            f"| {cid} | `{name}` | {ctype or 'TEXT'} "
                            f"| {'✓' if pk else ''} | {'✓' if notnull else ''} |"
                        )

                    lines.append("")
                    lines.append(sample_header.format(table_name=tbl))  # type: ignore[arg-type]
                    sample = conn.execute(f"SELECT * FROM '{tbl}' LIMIT {_SAMPLE_N}").fetchall()
                    if sample:
                        col_names = [c[1] for c in cols]
                        lines.append("```")
                        lines.append(", ".join(col_names))
                        for row in sample:
                            lines.append(", ".join(str(v) for v in row))
                        lines.append("```")
                    lines.append("")

                    # Merge TOML column descriptions if registered in databases.toml
                    toml_cols: dict[str, str] = get_table_meta(tbl).get("columns", {})
                    if toml_cols:
                        lines.append(col_desc_header.format(table_name=tbl))  # type: ignore[arg-type]
                        lines.append("")
                        for col_name, col_desc in toml_cols.items():
                            lines.append(f"- `{col_name}`: {col_desc}")
                        lines.append("")

        except sqlite3.Error as err:
            LOGGER.exception("SchemaChecker — failed to read %s", db_path)
            lines.append(schema_error.format(error=err))  # type: ignore[arg-type]

        return "\n".join(lines)

"""
genai/nl2sql_agent/sql_reviewer.py — Node 4 of the NL2SQL pipeline.

Validates the SQL statement produced by the SQL Writer before it reaches
the SQL Validator (Node 4.5) and the Executor.

ALL user-facing and reviewer feedback strings are loaded from
genai/prompt/sql_review.yaml — no message strings are hardcoded here.

Also contains RetryHandler (retry count management).

Four validation checks:
  1. Safety    — retry ceiling guard
  2. Safety    — disallowed DML/DDL keywords
  3. Structure — must start with SELECT and contain FROM
  4. Schema    — every referenced table must exist in the live database
"""

import re
import sqlite3
from pathlib import Path
from typing import Final

from genai.nl2sql_agent.state import AgentState
from utils.logger import get_logger
from utils.registry import get_prompt

LOGGER = get_logger("compliance_bot.nl2sql_agent.sql_reviewer")

_YAML = "sql_review.yaml"

_DISALLOWED_KEYWORDS: Final[frozenset[str]] = frozenset(
    {
        "INSERT", "UPDATE", "DELETE", "DROP", "TRUNCATE", "ALTER", "CREATE",
        "REPLACE", "PRAGMA", "ATTACH", "DETACH", "VACUUM",
    }
)

_FROM_PATTERN:  Final[re.Pattern[str]] = re.compile(r"\bFROM\b",  re.IGNORECASE)
_TABLE_PATTERN: Final[re.Pattern[str]] = re.compile(
    r"\bFROM\s+([`\"\[]?[\w]+[`\"\]]?)"
    r"|\bJOIN\s+([`\"\[]?[\w]+[`\"\]]?)",
    re.IGNORECASE,
)
_MAX_ALLOWED_RETRIES: Final[int] = 3



class RetryHandler:
    """Keeps retry count and enforces the ceiling."""

    def __init__(self, max_retries: int = _MAX_ALLOWED_RETRIES) -> None:
        self._max = max_retries

    @property
    def max_retries(self) -> int:
        return self._max

    def current_count(self, state: AgentState) -> int:
        return int(state.get("retry_count", 0))

    def should_retry(self, state: AgentState) -> bool:
        return self.current_count(state) < self._max

    def increment(self, state: AgentState) -> AgentState:
        n = self.current_count(state) + 1
        LOGGER.info("RetryHandler — retry_count → %d (max=%d)", n, self._max)
        return AgentState(retry_count=n)  # type: ignore[return-value]



class SQLReviewer:
    """
    LangGraph node — safety and schema validation on generated SQL.

    All feedback/error messages come from sql_review.yaml.
    max_retries — maximum writer → reviewer cycles before giving up.
    """

    def __init__(self, max_retries: int = _MAX_ALLOWED_RETRIES) -> None:
        self._max_retries = max_retries

    def run(self, state: AgentState) -> AgentState:
        """Validate the generated SQL through four sequential checks."""
        sql:         str = state.get("sql_query", "").strip()
        db_path_str: str = state.get("db_path", "")
        retry:       int = state.get("retry_count", 0)

        LOGGER.info("SQLReviewer — reviewing SQL (retry=%d): %s", retry, sql)

        # ── Guard: retry ceiling ───────────────────────────────────────────
        if retry >= self._max_retries:
            msg = get_prompt(
                _YAML, "feedback", "max_retries",
                default=(
                    f"⚠ The request could not be completed after {self._max_retries} attempts. "
                    "Please rephrase your question and try again."
                ),
            )
            msg = msg.format(max_retries=self._max_retries)  # type: ignore[union-attr]
            LOGGER.error("SQLReviewer — max retries exceeded.")
            return AgentState(  # type: ignore[return-value]
                sql_review_passed=False,
                review_feedback=msg,
                retry_count=retry,
                stages=["sql_reviewer:max_retries_exceeded"],
                error=msg,
            )

        # ── Guard: empty SQL ───────────────────────────────────────────────
        if not sql:
            return AgentState(  # type: ignore[return-value]
                sql_review_passed=False,
                review_feedback=get_prompt(
                    _YAML, "feedback", "empty_sql",
                    default="The SQL Writer returned an empty query.",
                ),
                retry_count=retry + 1,
                stages=["sql_reviewer:empty_sql"],
            )

        # ── Check 1: disallowed keywords ──────────────────────────────────
        upper_sql = sql.upper()
        for kw in _DISALLOWED_KEYWORDS:
            if re.search(rf"\b{kw}\b", upper_sql):
                LOGGER.warning("SQLReviewer — disallowed keyword: %s", kw)
                feedback_tpl = get_prompt(_YAML, "feedback", "disallowed_kw", default="")
                error_tpl    = get_prompt(_YAML, "user_error", "disallowed_kw", default="")
                return AgentState(  # type: ignore[return-value]
                    sql_review_passed=False,
                    review_feedback=(feedback_tpl or "").format(kw=kw),
                    retry_count=retry + 1,
                    stages=[f"sql_reviewer:disallowed_keyword={kw}"],
                    error=(error_tpl or "").format(kw=kw),
                )

        # ── Check 2: must start with SELECT ───────────────────────────────
        if not upper_sql.strip().startswith("SELECT"):
            LOGGER.warning("SQLReviewer — no SELECT prefix.")
            return AgentState(  # type: ignore[return-value]
                sql_review_passed=False,
                review_feedback=get_prompt(
                    _YAML, "feedback", "no_select",
                    default="The query must begin with SELECT.",
                ),
                retry_count=retry + 1,
                stages=["sql_reviewer:no_select"],
            )

        # ── Check 3: must contain FROM ────────────────────────────────────
        if not _FROM_PATTERN.search(sql):
            LOGGER.warning("SQLReviewer — missing FROM clause.")
            return AgentState(  # type: ignore[return-value]
                sql_review_passed=False,
                review_feedback=get_prompt(
                    _YAML, "feedback", "missing_from",
                    default="The query is missing a FROM clause.",
                ),
                retry_count=retry + 1,
                stages=["sql_reviewer:missing_from"],
            )

        # ── Check 4: all referenced tables must exist ──────────────────────
        if db_path_str:
            invalid = self._check_tables(sql, Path(db_path_str))
            if invalid:
                LOGGER.warning("SQLReviewer — unknown tables: %s", invalid)
                tpl = get_prompt(_YAML, "feedback", "unknown_tables", default="")
                return AgentState(  # type: ignore[return-value]
                    sql_review_passed=False,
                    review_feedback=(tpl or "").format(tables=", ".join(invalid)),
                    retry_count=retry + 1,
                    stages=[f"sql_reviewer:unknown_tables={invalid}"],
                )

        LOGGER.info("SQLReviewer — SQL passed all checks.")
        return AgentState(  # type: ignore[return-value]
            sql_review_passed=True,
            review_feedback=None,
            retry_count=retry,
            stages=["sql_reviewer:passed"],
        )

    @staticmethod
    def _check_tables(sql: str, db_path: Path) -> list[str]:
        """Return table names referenced in sql that don't exist in db_path."""
        if not db_path.exists():
            return []
        referenced: set[str] = set()
        for m in _TABLE_PATTERN.finditer(sql):
            name = (m.group(1) or m.group(2) or "").strip("`\"[]")
            if name:
                referenced.add(name.upper())
        if not referenced:
            return []
        try:
            with sqlite3.connect(str(db_path)) as conn:
                rows = conn.execute(
                    "SELECT UPPER(name) FROM sqlite_master WHERE type='table'"
                ).fetchall()
            existing: set[str] = {r[0] for r in rows}
        except sqlite3.Error as err:
            LOGGER.warning("SQLReviewer — can't read tables from %s: %s", db_path, err)
            return []
        return sorted(referenced - existing)

"""
genai/nl2sql_agent/sql_validator.py — Node 4.5 of the NL2SQL pipeline.

Uses sqlglot to parse the generated SQL at the AST level before it
reaches the database.  Catches syntax errors and non-SELECT statements
that slipped past the regex-based SQL Reviewer.

Three checks:
  1. Parse validity  — sqlglot must build an AST without errors.
  2. Statement type  — only SELECT statements are allowed.
  3. Dialect round-trip — canonicalise to SQLite syntax.

No prompt file needed — all output is structural pass/fail, no user messages.
"""


from genai.nl2sql_agent.state import AgentState
from utils.logger import get_logger

LOGGER = get_logger("compliance_bot.nl2sql_agent.sql_validator")

_ALLOWED_STMT_TYPES: frozenset[str] = frozenset({"select"})


class SQLValidator:
    """
    LangGraph node — AST-level SQL validation via sqlglot.

    dialect     — sqlglot dialect for parsing (default: "sqlite")
    strict_mode — reserved for future use
    """

    def __init__(self, dialect: str = "sqlite", strict_mode: bool = False) -> None:
        self._dialect     = dialect
        self._strict_mode = strict_mode

    def run(self, state: AgentState) -> AgentState:
        """Parse the SQL in state['sql_query'] and decide pass or retry."""
        sql:   str = state.get("sql_query", "").strip()
        retry: int = state.get("retry_count", 0)

        LOGGER.info("SQLValidator — validating: %s", sql)

        # If sqlglot isn't installed, skip and let the executor handle it.
        try:
            import sqlglot
            import sqlglot.errors
        except ImportError:
            LOGGER.warning("SQLValidator — sqlglot not installed; skipping AST check.")
            return AgentState(  # type: ignore[return-value]
                sqlglot_valid=True,
                sqlglot_error=None,
                stages=["sql_validator:skipped_no_sqlglot"],
            )

        # Check 1: parse.
        parse_error: str | None = None
        parsed = []
        try:
            parsed = sqlglot.parse(
                sql, dialect=self._dialect,
                error_level=sqlglot.errors.ErrorLevel.RAISE,
            )
        except sqlglot.errors.ParseError as exc:
            parse_error = str(exc)
            LOGGER.warning("SQLValidator — ParseError: %s", parse_error)

        if parse_error:
            return self._fail(
                retry, stage_tag="parse_error",
                reason=f"sqlglot could not parse the SQL: {parse_error}. Use standard SQLite syntax.",
            )

        if not parsed or parsed[0] is None:
            return self._fail(
                retry, stage_tag="empty_ast",
                reason="sqlglot returned an empty parse tree. Ensure the query is a valid SELECT.",
            )

        stmt = parsed[0]

        # Check 2: only SELECT allowed.
        stmt_type = type(stmt).__name__.lower()
        if stmt_type not in _ALLOWED_STMT_TYPES:
            LOGGER.warning("SQLValidator — non-SELECT type: %s", stmt_type)
            return self._fail(
                retry, stage_tag=f"disallowed_stmt_type={stmt_type}",
                reason=f"sqlglot identified this as '{stmt_type.upper()}'. Only SELECT is permitted.",
            )

        # Check 3: canonicalise to SQLite dialect.
        try:
            canonical: str = stmt.sql(dialect=self._dialect, pretty=False)
        except Exception:  # noqa: BLE001
            canonical = sql

        if canonical and canonical.strip() != sql.strip():
            LOGGER.info("SQLValidator — SQL normalised.\n  Before: %s\n  After : %s", sql, canonical)

        LOGGER.info("SQLValidator — passed AST validation.")
        return AgentState(  # type: ignore[return-value]
            sqlglot_valid=True,
            sqlglot_error=None,
            sql_query=canonical or sql,
            stages=[f"sql_validator:passed dialect={self._dialect}"],
        )

    @staticmethod
    def _fail(retry: int, reason: str, stage_tag: str) -> AgentState:
        """Return a partial AgentState marking validation as failed."""
        return AgentState(  # type: ignore[return-value]
            sqlglot_valid=False,
            sqlglot_error=reason,
            sql_review_passed=False,
            review_feedback=reason,
            retry_count=retry + 1,
            stages=[f"sql_validator:{stage_tag}"],
        )

_INLINE_COMMENT_RE: re.Pattern[str] = re.compile(r"--[^\n]*", re.MULTILINE)
_BLOCK_COMMENT_RE:  re.Pattern[str] = re.compile(r"/\*.*?\*/", re.DOTALL)
_WHITESPACE_RE:     re.Pattern[str] = re.compile(r"[ \t]+")


def strip_comments(sql: str) -> str:
    """Remove all -- inline and /* */ block comments from sql."""
    sql = _BLOCK_COMMENT_RE.sub(" ", sql)
    return _INLINE_COMMENT_RE.sub("", sql).strip()


def normalise_whitespace(sql: str) -> str:
    """Collapse horizontal whitespace runs to a single space on each line."""
    lines = [_WHITESPACE_RE.sub(" ", line).strip() for line in sql.splitlines()]
    return "\n".join(line for line in lines if line)


def strip_trailing_semicolon(sql: str) -> str:
    """Remove trailing semicolons and whitespace."""
    return sql.rstrip("; \t\n")


def optimise(sql: str) -> str:
    """
    Strip comments → normalise whitespace → remove trailing semicolons.
    Never changes query semantics.
    """
    sql = strip_comments(sql)
    sql = normalise_whitespace(sql)
    sql = strip_trailing_semicolon(sql)
    LOGGER.debug("optimise — result: %s", sql[:200])
    return sql

class ContextBuilder:
    """
    Builds the full LLM prompt for the SQL Writer Agent.

    System role, rules, and template are loaded from sql_writer.yaml.
    Table metadata and few-shot examples come from databases.toml via registry.
    Zero hardcoded strings — edit the YAML files to change prompt wording.
    """

    def __init__(self, table_name: str) -> None:
        self._table_name = table_name

    def build(
        self,
        question: str,
        schema_context: str,
        feedback: str | None = None,
    ) -> str:
        """
        Assemble and return the full prompt string.

        question       — normalised question from IntentChecker
        schema_context — Markdown schema block from SchemaChecker
        feedback       — reviewer feedback on retry path (None on first attempt)
        """
        meta       = get_table_meta(self._table_name)
        table_desc = meta.get("description", "")
        col_desc: dict[str, str] = meta.get("columns", {})
        few_shots  = get_few_shots(self._table_name)

        # Format column block and rules
        column_block = "\n".join(
            f"  - **{col}**: {desc}" for col, desc in col_desc.items()
        )
        rules_block = "\n".join(
            f"  {i+1}. {rule.format(begin_tag=SQL_BEGIN_TAG, end_tag=SQL_END_TAG)}"
            for i, rule in enumerate(_RULES)
        )
        retry_section = (
            f"\n{_RETRY_HEADER}\n{feedback}\n" if feedback else ""
        )

        LOGGER.debug(
            "ContextBuilder — table=%s question=%r retry=%s",
            self._table_name, question[:80], bool(feedback),
        )

        # Use the TOML template when available; fall back to inline format
        if _TEMPLATE:
            return _TEMPLATE.format(
                system_role=_SYSTEM_ROLE,
                rules_block=rules_block,
                table_name=self._table_name,
                table_description=table_desc,
                column_block=column_block,
                schema_context=schema_context,
                few_shots=few_shots,
                retry_section=retry_section,
                question=question,
            )

        # Fallback (shouldn't normally be reached)
        return (
            f"[INST] <<SYS>>\n{_SYSTEM_ROLE}\n\nRules:\n{rules_block}\n<</SYS>>\n\n"
            f"=== Table Metadata: {self._table_name} ===\n"
            f"Description: {table_desc}\n\nColumns:\n{column_block}\n\n"
            f"=== Live Database Schema ===\n{schema_context}\n\n"
            f"=== Few-shot Examples ===\n{few_shots}\n"
            f"{retry_section}\n"
            f"Question: {question}\nResponse:\n[/INST]"
        )



class SQLWriterAgent:
    """
    LangGraph node — calls the LLM to generate a SQLite SELECT statement.

    Prompt is built by ContextBuilder (YAML-driven, no hardcoded strings).
    Table metadata comes from databases.toml — add new databases/tables there.
    """

    def __init__(
        self,
        model_service: ModelLoaderService,
        table_name: str = COMPLIANCE_TABLE_NAME,
    ) -> None:
        self._model_service  = model_service
        self._context_builder = ContextBuilder(table_name=table_name)

    def run(self, state: AgentState) -> AgentState:
        """Generate a SQL SELECT query for the current pipeline state."""
        question: str      = state.get("interpreted_question", state.get("question", ""))
        schema_context: str = state.get("schema_context", "")
        feedback: str | None = state.get("review_feedback")
        retry: int         = state.get("retry_count", 0)

        LOGGER.info("SQLWriterAgent — generating SQL (retry=%d) for: %r", retry, question)

        prompt     = self._context_builder.build(question, schema_context, feedback)
        raw_output = self._model_service.generate(prompt=prompt)
        sql        = self._model_service.extract_sql(raw_output)

        if not sql:
            LOGGER.warning("SQLWriterAgent — no SQL tags in output. Raw: %.400s", raw_output)
            return AgentState(  # type: ignore[return-value]
                sql_query="",
                retry_count=retry,
                stages=["sql_writer_agent:extraction_failed"],
                error=f"LLM did not return SQL within expected delimiters. Raw: {raw_output[:300]}",
            )

        LOGGER.info("SQLWriterAgent — LLM SQL: %s", sql)
        return AgentState(  # type: ignore[return-value]
            sql_query=sql,
            retry_count=retry,
            stages=[f"sql_writer_agent:llm_generated retry={retry}"],
        )

"""
Shared LangGraph state definition for the NL2SQL agentic pipeline.

``AgentState`` is the single mutable dict that flows through every node in the
compiled StateGraph.  LangGraph merges partial updates automatically, so each
node only needs to return the keys it owns.
"""

import operator
from typing import Annotated

from typing_extensions import TypedDict


class AgentState(TypedDict, total=False):
    """
    Pipeline-wide state threaded through all five NL2SQL agent nodes.

    Fields are ``total=False`` so each node writes only the keys it owns.
    The ``stages`` list uses an ``Annotated`` reducer so LangGraph appends
    entries from each node rather than overwriting the whole list.
    """

    # Input
    question: str
    question_id: str | None

    # IntentAgent outputs
    interpreted_question: str
    intent_ok: bool
    clarification_needed: str | None

    # SchemaAgent outputs
    db_path: str
    schema_context: str

    # SQLWriterAgent outputs
    sql_query: str

    # SQLReviewAgent outputs
    sql_review_passed: bool
    review_feedback: str | None
    retry_count: int

    # SQLValidatorAgent outputs (sqlglot AST parse + dialect check)
    sqlglot_valid: bool
    sqlglot_error: str | None

    # ExecutorAgent outputs
    result_rows: list[dict[str, object]]
    result_columns: list[str]
    result_summary: str

    # Shared bookkeeping
    error: str | None
    stages: Annotated[list[str], operator.add]

"""
utils/helpers.py — stateless utility helpers.

Merges:
  - common.py    (safe_json_loads, truncate, normalise_whitespace, flatten_list)
  - file_utils.py (ensure_directory, read/write text/bytes, safe_delete, list_files …)
  - json_utils.py (safe_loads, safe_dumps, extract_json_from_text, …)

Nothing in this file has side-effects or external dependencies beyond the
Python standard library.
"""

import json
import re
import shutil
from pathlib import Path
from typing import Any


def safe_json_loads(text: str) -> dict[str, Any] | None:
    """Try to parse text as JSON; return None if parsing fails."""
    try:
        return json.loads(text)
    except (json.JSONDecodeError, TypeError):
        return None


def truncate(text: str, max_len: int = 300, suffix: str = "…") -> str:
    """Shorten text to max_len characters and append suffix if it was cut."""
    if len(text) <= max_len:
        return text
    return text[: max_len - len(suffix)] + suffix


def normalise_whitespace(text: str) -> str:
    """Collapse any run of whitespace (spaces, tabs, newlines) into a single space."""
    return re.sub(r"\s+", " ", text).strip()


def flatten_list(nested: list[Any]) -> list[Any]:
    """
    Flatten exactly one level of nesting in a list.

    Example: [[1, 2], [3], 4] → [1, 2, 3, 4]
    """
    result: list[Any] = []
    for item in nested:
        if isinstance(item, list):
            result.extend(item)
        else:
            result.append(item)
    return result


def ensure_directory(path: str | Path) -> Path:
    """Create the directory (and any missing parents) if it doesn't exist yet."""
    resolved = Path(path).resolve()
    resolved.mkdir(parents=True, exist_ok=True)
    return resolved


def read_text(path: str | Path, encoding: str = "utf-8") -> str:
    """Read a text file and return its contents as a string."""
    return Path(path).read_text(encoding=encoding)


def write_text(path: str | Path, content: str, encoding: str = "utf-8") -> Path:
    """Write content to a file, creating parent directories if needed."""
    dest = Path(path).resolve()
    dest.parent.mkdir(parents=True, exist_ok=True)
    dest.write_text(content, encoding=encoding)
    return dest


def read_bytes(path: str | Path) -> bytes:
    """Read a file and return its raw bytes."""
    return Path(path).read_bytes()


def write_bytes(path: str | Path, data: bytes) -> Path:
    """Write binary data to a file, creating parent directories if needed."""
    dest = Path(path).resolve()
    dest.parent.mkdir(parents=True, exist_ok=True)
    dest.write_bytes(data)
    return dest


def safe_delete(path: str | Path) -> bool:
    """
    Delete a file or entire directory tree without raising if it doesn't exist.

    Returns True if something was deleted, False if the path was already gone.
    """
    target = Path(path)
    if not target.exists():
        return False
    if target.is_dir():
        shutil.rmtree(target)
    else:
        target.unlink()
    return True


def list_files(
    directory: str | Path,
    pattern: str = "*",
    recursive: bool = False,
) -> list[Path]:
    """
    Return a sorted list of files matching pattern inside directory.

    Set recursive=True to search all subdirectories as well.
    """
    root = Path(directory)
    globber = root.rglob if recursive else root.glob
    return sorted(p for p in globber(pattern) if p.is_file())


def file_size(path: str | Path) -> int:
    """Return the byte size of a file, or 0 if the file does not exist."""
    target = Path(path)
    return target.stat().st_size if target.exists() else 0


def resolve_project_path(*parts: str) -> Path:
    """
    Build an absolute path relative to the project root.

    The project root is one level above utils/ (workspace root).
    Example: resolve_project_path("data", "comp_db", "COMP_REPORTS.db")
    """
    project_root = Path(__file__).resolve().parents[1]
    return project_root.joinpath(*parts)


_JSON_OBJECT_PATTERN: re.Pattern[str] = re.compile(r"(\{.*?\}|\[.*?\])", re.DOTALL)


def safe_loads(text: str) -> Any:
    """Parse text as JSON; return None instead of raising on bad input."""
    try:
        return json.loads(text)
    except (json.JSONDecodeError, TypeError):
        return None


def safe_dumps(obj: Any, indent: int = 2, default: Any = str) -> str:
    """Serialise obj to JSON, converting un-serialisable values with str()."""
    return json.dumps(obj, indent=indent, default=default, ensure_ascii=False)


def extract_json_from_text(text: str) -> Any:
    """
    Scan free-form text for the first valid JSON object or array and return it.

    Useful when the LLM wraps its JSON answer in explanatory prose.
    Returns None if no valid JSON block is found.
    """
    for match in _JSON_OBJECT_PATTERN.finditer(text):
        parsed = safe_loads(match.group(0))
        if parsed is not None:
            return parsed
    return None


def extract_json_array(text: str) -> list[Any]:
    """Return the first JSON array found in text, or [] if none exists."""
    result = extract_json_from_text(text)
    if isinstance(result, list):
        return result
    return []


def extract_json_object(text: str) -> dict[str, Any]:
    """Return the first JSON object found in text, or {} if none exists."""
    result = extract_json_from_text(text)
    if isinstance(result, dict):
        return result
    return {}


def pretty_print(obj: Any) -> str:
    """Return a nicely indented JSON string for obj."""
    return safe_dumps(obj, indent=2)


def is_valid_json(text: str) -> bool:
    """Return True if text is syntactically valid JSON, False otherwise."""
    return safe_loads(text) is not None

"""
Centralised logging factory for the Compliance Chatbot.

All modules must obtain their logger exclusively through :func:`get_logger`
to guarantee consistent formatting and handler configuration.

Configuration is loaded from ``genai/config/logging.toml`` (TOML-based,
Python 3.11+ tomllib).  Falls back to a minimal ``basicConfig`` when the
TOML file is absent or unreadable.
"""

import logging
import logging.config
import logging.handlers
from pathlib import Path

_LOGGING_CONFIGURED: bool = False

_PROJECT_ROOT: Path = Path(__file__).resolve().parents[1]
_TOML_CONFIG_PATH: Path = _PROJECT_ROOT / "genai" / "config" / "logging.toml"


def _load_toml_logging_config(path: Path) -> dict | None:
    """Load logging configuration from a TOML file. Returns None on failure."""
    try:
        try:
            import tomllib  # Python 3.11+
        except ImportError:
            try:
                import tomli as tomllib  # type: ignore[no-redef]
            except ImportError:
                return None

        with path.open("rb") as fh:
            data = tomllib.load(fh)
        return data.get("logging")
    except Exception:  # noqa: BLE001
        return None


def _configure_logging(log_dir: str | None = None) -> None:
    """
    Bootstrap logging from ``genai/config/logging.toml``.

    Falls back to a minimal ``basicConfig`` if the TOML file is absent or
    the tomllib/tomli package is not available.

    Args:
        log_dir: Optional override for the log output directory.
    """
    global _LOGGING_CONFIGURED  # pylint: disable=global-statement

    if _LOGGING_CONFIGURED:
        return

    # Resolve log directory against project root so it is always
    # <project_root>/data/logs regardless of the process CWD.
    raw_dir = Path(log_dir) if log_dir else Path("data/logs")
    resolved_log_dir = raw_dir if raw_dir.is_absolute() else _PROJECT_ROOT / raw_dir
    resolved_log_dir.mkdir(parents=True, exist_ok=True)

    log_config = _load_toml_logging_config(_TOML_CONFIG_PATH)

    if log_config:
        # Patch the rotating_file handler filename if a custom log_dir was given.
        handlers = log_config.get("handlers", {})
        rotating = handlers.get("rotating_file")
        if rotating:
            rotating["filename"] = str(resolved_log_dir / "compliance_bot.log")

        try:
            logging.config.dictConfig(log_config)
            _LOGGING_CONFIGURED = True
            return
        except Exception:  # noqa: BLE001
            logging.getLogger(__name__).debug(
                "dictConfig failed — falling back to basicConfig."
            )

    # Fallback: simple basicConfig.
    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s | %(levelname)-8s | %(name)s | %(message)s",
    )
    logging.getLogger(__name__).warning(
        "Logging TOML config not found or unreadable at %s — using basicConfig fallback.",
        _TOML_CONFIG_PATH,
    )

    _LOGGING_CONFIGURED = True


def get_logger(name: str, log_dir: str | None = None) -> logging.Logger:
    """
    Return a named logger, ensuring logging is configured beforehand.

    Args:
        name:    Dot-separated logger name (e.g. ``"compliance_bot.core"``).
        log_dir: Optional log output directory; used only on first call.

    Returns:
        Configured :class:`logging.Logger` instance.
    """
    _configure_logging(log_dir=log_dir)
    return logging.getLogger(name)

"""
utils/registry.py — data-access registries: databases, prompts, SQLite helpers.

Merges:
  - db_registry.py   — databases.toml metadata registry (tables, columns, few-shots)
  - db_utils.py      — read-only SQLite connection + query helpers
  - prompt_loader.py — YAML prompt file loader with in-memory cache

All three share the same concern: reading persistent data stores (TOML files
and SQLite databases) and caching the results in memory.
"""

import contextlib
import sqlite3
from collections.abc import Generator
from contextlib import contextmanager
from pathlib import Path
from typing import Any

import yaml

from utils.logger import get_logger

LOGGER = get_logger("compliance_bot.utils.registry")


def _load_toml_file(path: Path) -> dict[str, Any]:
    """Open and parse a TOML file; return {} on any failure."""
    try:
        try:
            import tomllib  # built-in Python 3.11+
        except ImportError:
            import tomli as tomllib  # type: ignore[no-redef]
        with path.open("rb") as fh:
            return tomllib.load(fh)
    except Exception as exc:  # noqa: BLE001
        LOGGER.warning("registry._load_toml_file — failed to load %s: %s", path, exc)
        return {}


_DATABASES_TOML = Path(__file__).resolve().parents[1] / "genai" / "config" / "databases.toml"

_db_registry: dict[str, Any] | None = None


def _build_db_registry() -> dict[str, Any]:
    """Parse databases.toml and return a flat {table_name: metadata} dict."""
    if not _DATABASES_TOML.exists():
        LOGGER.warning("registry: databases.toml not found at %s", _DATABASES_TOML)
        return {}

    raw = _load_toml_file(_DATABASES_TOML)
    registry: dict[str, Any] = {}

    for db_stem, db_data in raw.get("databases", {}).items():
        db_file = db_data.get("db_file", "")
        for table_name, table_data in db_data.get("tables", {}).items():
            few_shots_raw = table_data.get("few_shots", {})
            few_shots_str = (
                few_shots_raw.get("examples", "").strip()
                if isinstance(few_shots_raw, dict)
                else ""
            )
            registry[table_name] = {
                "db_stem":      db_stem,
                "db_file":      db_file,
                "description":  table_data.get("description", ""),
                "filter_query": table_data.get("filter_query", ""),
                "columns":      table_data.get("columns", {}),
                "few_shots":    few_shots_str,
            }
            LOGGER.debug("registry: registered table %s (db=%s)", table_name, db_stem)

    LOGGER.info("registry: loaded %d table(s) from databases.toml", len(registry))
    return registry


def _get_db_registry() -> dict[str, Any]:
    """Return the cached registry, building it on first call."""
    global _db_registry
    if _db_registry is None:
        _db_registry = _build_db_registry()
    return _db_registry


def reload() -> None:
    """Clear the database registry cache so the next access re-reads databases.toml."""
    global _db_registry
    _db_registry = None
    LOGGER.info("registry: db cache cleared — will reload on next access.")


def get_table_meta(table_name: str) -> dict[str, Any]:
    """Return the full metadata dict for a registered table, or {} if unknown."""
    return _get_db_registry().get(table_name, {})


def get_few_shots(table_name: str) -> str:
    """Return the few-shot example block for a table, or '' if not registered."""
    return _get_db_registry().get(table_name, {}).get("few_shots", "")


def get_all_tables() -> list[str]:
    """Return a sorted list of every registered table name."""
    return sorted(_get_db_registry().keys())


def list_databases() -> list[str]:
    """Return a sorted list of every registered database stem name."""
    reg = _get_db_registry()
    return sorted({meta["db_stem"] for meta in reg.values()})


@contextmanager
def open_readonly(db_path: str | Path) -> Generator[sqlite3.Connection, None, None]:
    """
    Context manager: open a SQLite file in deferred read-only mode.

    Always rolls back and closes the connection on exit — even on error.
    Usage:
        with open_readonly("data/comp_db/COMP_REPORTS.db") as conn:
            rows = conn.execute("SELECT COUNT(*) FROM COMP_TBL").fetchone()
    """
    path = Path(db_path)
    conn: sqlite3.Connection | None = None
    try:
        conn = sqlite3.connect(str(path), check_same_thread=False)
        conn.row_factory = sqlite3.Row
        conn.execute("BEGIN DEFERRED")
        yield conn
    finally:
        if conn is not None:
            with contextlib.suppress(sqlite3.Error):
                conn.rollback()
            conn.close()


def fetch_all(
    db_path: str | Path,
    sql: str,
    params: tuple[Any, ...] = (),
) -> list[dict[str, Any]]:
    """Run sql and return every result row as a list of {column: value} dicts."""
    with open_readonly(db_path) as conn:
        cursor = conn.execute(sql, params)
        col_names: list[str] = [d[0] for d in (cursor.description or [])]
        return [dict(zip(col_names, row, strict=False)) for row in cursor.fetchall()]


def fetch_scalar(
    db_path: str | Path,
    sql: str,
    params: tuple[Any, ...] = (),
    default: Any = None,
) -> Any:
    """Run sql and return just the first column of the first row (e.g. COUNT(*))."""
    with open_readonly(db_path) as conn:
        row = conn.execute(sql, params).fetchone()
    return row[0] if row else default


def list_tables(db_path: str | Path) -> list[str]:
    """Return a sorted list of all table names in the database."""
    sql = "SELECT name FROM sqlite_master WHERE type='table' ORDER BY name"
    with open_readonly(db_path) as conn:
        rows = conn.execute(sql).fetchall()
    return [row[0] for row in rows]


def table_exists(db_path: str | Path, table_name: str) -> bool:
    """Return True if table_name exists in the database (case-insensitive check)."""
    sql = "SELECT 1 FROM sqlite_master WHERE type='table' AND UPPER(name)=UPPER(?)"
    return fetch_scalar(db_path, sql, (table_name,)) is not None


def get_column_names(db_path: str | Path, table_name: str) -> list[str]:
    """Return the ordered column names for a table via PRAGMA table_info."""
    with open_readonly(db_path) as conn:
        rows = conn.execute(f"PRAGMA table_info('{table_name}')").fetchall()
    return [row[1] for row in rows]


def row_count(db_path: str | Path, table_name: str) -> int:
    """Return how many rows are in the table; returns 0 on any error."""
    try:
        return fetch_scalar(db_path, f"SELECT COUNT(*) FROM '{table_name}'", default=0)  # noqa: S608
    except sqlite3.Error as err:
        LOGGER.warning("registry.row_count — error on %s.%s: %s", db_path, table_name, err)
        return 0


_PROMPTS_DIR: Path = Path(__file__).resolve().parents[1] / "genai" / "prompt"

_prompt_cache: dict[str, dict[str, Any]] = {}


def _load_yaml_file(path: Path) -> dict[str, Any]:
    """Open and parse a YAML file; return {} on any failure."""
    try:
        with path.open("r", encoding="utf-8") as fh:
            return yaml.safe_load(fh) or {}
    except Exception as exc:  # noqa: BLE001
        LOGGER.warning("registry._load_yaml_file — failed to load %s: %s", path, exc)
        return {}


def load_prompt(filename: str) -> dict[str, Any]:
    """
    Load and cache a YAML prompt file from genai/prompt/.

    filename — e.g. "sql_writer.yaml" (or bare stem "sql_writer")
    Returns the full parsed dict, or {} on error.
    Raises FileNotFoundError if the file does not exist.
    """
    stem = filename.removesuffix(".yaml").removesuffix(".toml")

    if stem in _prompt_cache:
        return _prompt_cache[stem]

    path = _PROMPTS_DIR / (stem + ".yaml")
    if not path.exists():
        raise FileNotFoundError(f"Prompt file not found: {path}")

    data = _load_yaml_file(path)
    _prompt_cache[stem] = data
    LOGGER.debug("registry: loaded prompt file: %s", path)
    return data


def get_prompt(filename: str, *keys: str, default: str | None = None) -> str | None:
    """
    Retrieve a nested string value from a YAML prompt file.

    Example: get_prompt("intent_checker.yaml", "messages", "clarification")

    Walks the key path and returns the string at that location,
    or default if any key is missing.
    """
    data = load_prompt(filename)
    node: Any = data
    for key in keys:
        if not isinstance(node, dict):
            return default
        node = node.get(key)
        if node is None:
            return default
    return str(node) if node is not None else default
